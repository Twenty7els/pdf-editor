    34→function hexToRgb(hex: string): { r: number; g: number; b: number } | null {
    35→  const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
    36→  return result
    37→    ? {
    38→        r: parseInt(result[1], 16) / 255,
    39→        g: parseInt(result[2], 16) / 255,
    40→        b: parseInt(result[3], 16) / 255,
    41→      }
    42→    : null;
    43→}

   163→  const {
   164→    pdfFile,

   166→    stamps,
   167→    texts,

   201→  const handleFileChange = useCallback(
   202→    (e: React.ChangeEvent<HTMLInputElement>) => {
   203→      const file = e.target.files?.[0];
   204→      if (file && file.type === "application/pdf") {
   205→        setPdfFile(file);
   206→        toast.success("PDF загружен", {
   207→          description: file.name,
   208→        });
   209→      } else {
   210→        toast.error("Пожалуйста, выберите PDF файл");
   211→      }
   212→      if (fileInputRef.current) {
   213→        fileInputRef.current.value = "";
   214→      }
   215→    },
   216→    [setPdfFile]
   217→  );
   218→
   219→  const handleDownload = useCallback(async () => {
   220→    if (!pdfArrayBuffer || isDownloading) return;
   221→
   222→    setIsDownloading(true);
   223→    const loadingToast = toast.loading("Подготовка PDF...");
   224→
   225→    try {
   226→      const { PDFDocument, rgb, degrees } = await import("pdf-lib");
   227→
   228→      const pdfDoc = await PDFDocument.load(pdfArrayBuffer);
   229→
   230→      const fontkit = (await import("@pdf-lib/fontkit")).default;
   231→      pdfDoc.registerFontkit(fontkit);
   232→
   233→      // Read document-level state from store
   234→      const state = usePdfEditorStore.getState();
   235→      const { pageRotations, deletedPages, exportPageSelection } = state;
   236→
   237→      // Determine which pages (1-indexed) to keep in the export
   238→      const pageCount = pdfDoc.getPageCount();
   239→      const pagesToKeep: number[] = [];
   240→      for (let p = 1; p <= pageCount; p++) {
   241→        if (deletedPages.includes(p)) continue;
   242→        if (exportPageSelection !== null && !exportPageSelection.includes(p))
   243→          continue;
   244→        pagesToKeep.push(p);
   245→      }
   246→      const keepSet = new Set(pagesToKeep);
   247→
   248→      // Apply page rotations to kept pages BEFORE drawing items
   249→      for (const pageNum of pagesToKeep) {
   250→        const userRotation = pageRotations[pageNum] || 0;
   251→        if (userRotation !== 0) {
   252→          const page = pdfDoc.getPage(pageNum - 1);
   253→          const existing = page.getRotation().angle || 0;
   254→          page.setRotation(degrees((existing + userRotation) % 360));
   255→        }
   256→      }
   257→
   258→      let unicodeFonts: {
   259→        regular: unknown;
   260→        bold: unknown;
   261→        italic: unknown;
   262→        boldItalic: unknown;
   263→      } | null = null;
   264→      const getUnicodeFonts = async () => {
   265→        if (unicodeFonts) return unicodeFonts;
   266→        const {
   267→          NOTO_SANS_REGULAR,
   268→          NOTO_SANS_BOLD,
   269→          NOTO_SANS_ITALIC,
   270→          NOTO_SANS_BOLD_ITALIC,
   271→        } = await import("@/lib/font-base64");
   272→        const decode = (b64: string) =>
   273→          new Uint8Array(atob(b64).split("").map((c) => c.charCodeAt(0)));
   274→        unicodeFonts = {
   275→          regular: await pdfDoc.embedFont(decode(NOTO_SANS_REGULAR)),
   276→          bold: await pdfDoc.embedFont(decode(NOTO_SANS_BOLD)),
   277→          italic: await pdfDoc.embedFont(decode(NOTO_SANS_ITALIC)),
   278→          boldItalic: await pdfDoc.embedFont(decode(NOTO_SANS_BOLD_ITALIC)),
   279→        };
   280→        return unicodeFonts;
   281→      };
   282→
   283→      // Stamps — skip if page not in keepSet
   284→      for (const stamp of stamps.filter(
   285→        (s) => !s.hidden && keepSet.has(s.page)
   286→      )) {
   287→        try {
   288→          const page = pdfDoc.getPage(stamp.page - 1);
   289→          const { width: pageWidth, height: pageHeight } = page.getSize();
   290→
   291→          const cw = stamp.canvasWidth || 800;
   292→          const ch = stamp.canvasHeight || 1100;
   293→
   294→          const pdfX = (stamp.x / cw) * pageWidth;
   295→          const pdfY = pageHeight - ((stamp.y + stamp.height) / ch) * pageHeight;
   296→          const pdfWidth = (stamp.width / cw) * pageWidth;
   297→          const pdfHeight = (stamp.height / ch) * pageHeight;
   298→
   299→          // Compensate rotation: pdf-lib rotates around (x, y) = bottom-left corner,
   300→          // but canvas rotates around center. Adjust (x, y) so center stays in place.
   301→          const rad = (stamp.rotation * Math.PI) / 180;
   302→          const cx = pdfX + pdfWidth / 2;
   303→          const cy = pdfY + pdfHeight / 2;
   304→          const adjX = cx - (pdfWidth / 2) * Math.cos(rad) + (pdfHeight / 2) * Math.sin(rad);
   305→          const adjY = cy - (pdfWidth / 2) * Math.sin(rad) - (pdfHeight / 2) * Math.cos(rad);
   306→
   307→          const response = await fetch(stamp.src);
   308→          const imageArrayBuffer = await response.arrayBuffer();
   309→          const imageBytes = new Uint8Array(imageArrayBuffer);
   310→
   311→          let image;
   312→          try {
   313→            image = await pdfDoc.embedPng(imageBytes);
   314→          } catch {
   315→            try {
   316→              image = await pdfDoc.embedJpg(imageBytes);
   317→            } catch {
   318→              console.error("Could not embed stamp image, skipping");
   319→              continue;
   320→            }
   321→          }
   322→
   323→          page.drawImage(image, {
   324→            x: adjX,
   325→            y: adjY,
   326→            width: pdfWidth,
   327→            height: pdfHeight,
   328→            rotate: degrees(stamp.rotation),
   329→            opacity: stamp.opacity,
   330→          });
   331→        } catch (err) {
   332→          console.error("Error embedding stamp:", err);
   333→        }
   334→      }
   335→
   336→      // Texts — skip if page not in keepSet
   337→      for (const textItem of texts.filter(
   338→        (t) => !t.hidden && keepSet.has(t.page)
   339→      )) {
   340→        if (!textItem.text.trim()) continue;
   341→
   342→        try {
   343→          const page = pdfDoc.getPage(textItem.page - 1);
   344→          const { width: pageWidth, height: pageHeight } = page.getSize();
   345→
   346→          const cw = textItem.canvasWidth || 800;
   347→          const ch = textItem.canvasHeight || 1100;
   348→
   349→          const scaledFontSize = (textItem.fontSize / ch) * pageHeight;
   350→          // letterSpacing in canvas px → PDF points (text space)
   351→          const scaledLetterSpacing =
   352→            (textItem.letterSpacing / ch) * pageHeight;
   353→
   354→          let font: import("pdf-lib").PDFFont;
   355→          const uFonts = await getUnicodeFonts();
   356→          if (textItem.bold && textItem.italic) font = uFonts.boldItalic as import("pdf-lib").PDFFont;
   357→          else if (textItem.bold) font = uFonts.bold as import("pdf-lib").PDFFont;
   358→          else if (textItem.italic) font = uFonts.italic as import("pdf-lib").PDFFont;
   359→          else font = uFonts.regular as import("pdf-lib").PDFFont;
   360→
   361→          const color = hexToRgb(textItem.color);
   362→          const pdfColor = color
   363→            ? rgb(color.r, color.g, color.b)
   364→            : rgb(0, 0, 0);
   365→
   366→          // Multiline support
   367→          const lines = textItem.text.split("\n");
   368→
   369→          // Font ascent — distance from baseline to top of glyph.
   370→          // CSS positions text by top of line-box; glyph is centered within line-height.
   371→          // PDF positions text by baseline. So: baseline = top + leading/2 + ascent.
   372→          const fontAscent = font.heightAtSize(scaledFontSize, {
   373→            descender: false,
   374→          });
   375→          // Line height = 1.2 * fontSize (matches canvas CSS lineHeight: 1.2)
   376→          const scaledLineHeight = scaledFontSize * 1.2;
   377→          // Leading is distributed equally above and below the glyph in CSS
   378→          const leadingHalf = (scaledLineHeight - scaledFontSize) / 2;
   379→
   380→          for (let li = 0; li < lines.length; li++) {
   381→            const line = lines[li];
   382→            if (!line) {
   383→              continue;
   384→            }
   385→
   386→            // Canvas: top of line-box = textItem.y + li * (fontSize * 1.2)
   387→            // Glyph top = top of line-box + leading/2
   388→            // PDF baseline = glyph top + ascent
   389→            const lineCanvasTop = textItem.y + li * textItem.fontSize * 1.2;
   390→            const lineBaselinePdfY =
   391→              pageHeight -
   392→              ((lineCanvasTop + leadingHalf + fontAscent) / ch) * pageHeight;
   393→
   394→            // Measure line width for alignment
   395→            const lineWidth = font.widthOfTextAtSize(line, scaledFontSize);
   396→
   397→            // Compute X based on alignment
   398→            let linePdfX: number;
   399→            const basePdfX = (textItem.x / cw) * pageWidth;
   400→            if (textItem.align === "center") {
   401→              linePdfX = basePdfX - lineWidth / 2;
   402→            } else if (textItem.align === "right") {
   403→              linePdfX = basePdfX - lineWidth;
   404→            } else {
   405→              linePdfX = basePdfX;
   406→            }
   407→
   408→            // Compensate rotation for text (pdf-lib rotates around baseline-left,
   409→            // canvas rotates around center). Adjust (x, y) so center stays in place.
   410→            const textRad = (textItem.rotation * Math.PI) / 180;
   411→            const lineCenterX = linePdfX + lineWidth / 2;
   412→            const lineCenterY = lineBaselinePdfY + scaledFontSize / 2;
   413→            const textAdjX =
   414→              lineCenterX -
   415→              (lineWidth / 2) * Math.cos(textRad) +
   416→              (scaledFontSize / 2) * Math.sin(textRad);
   417→            const textAdjY =
   418→              lineCenterY -
   419→              (lineWidth / 2) * Math.sin(textRad) -
   420→              (scaledFontSize / 2) * Math.cos(textRad);
   421→
   422→            page.drawText(line, {
   423→              x: textAdjX,
   424→              y: textAdjY,
   425→              size: scaledFontSize,
   426→              font,
   427→              color: pdfColor,
   428→              rotate: degrees(textItem.rotation),
   429→              // charSpacing is in text space units (1/1000 of font size)
   430→              charSpacing: (scaledLetterSpacing / scaledFontSize) * 1000,
   431→            });
   432→
   433→            // Underline: draw a thin rectangle below the baseline
   434→            if (textItem.underline) {
   435→              const underlineThickness = Math.max(
   436→                0.5,
   437→                scaledFontSize / 18
   438→              );
   439→              const underlineY = textAdjY - scaledFontSize * 0.12;
   440→              page.drawRectangle({
   441→                x: textAdjX,
   442→                y: underlineY,
   443→                width: lineWidth,
   444→                height: underlineThickness,
   445→                color: pdfColor,
   446→                rotate: degrees(textItem.rotation),
   447→              });
   448→            }
   449→          }
   450→        } catch (err) {
   451→          console.error("Error drawing text:", err);
   452→        }
   453→      }
   454→
   455→      // Erasers — skip if page not in keepSet
   456→      for (const eraserItem of erasers.filter(
   457→        (e) => !e.hidden && keepSet.has(e.page)
   458→      )) {
   459→        try {
   460→          if (eraserItem.points.length === 0) continue;
   461→
   462→          const page = pdfDoc.getPage(eraserItem.page - 1);
   463→          const { width: pageWidth, height: pageHeight } = page.getSize();
   464→
   465→          const cw = eraserItem.canvasWidth || 800;
   466→          const ch = eraserItem.canvasHeight || 1100;
   467→
   468→          const eraserColor = hexToRgb(eraserItem.color);
   469→          const pdfColor = eraserColor
   470→            ? rgb(eraserColor.r, eraserColor.g, eraserColor.b)
   471→            : rgb(1, 1, 1);
   472→
   473→          const pdfStrokeWidth = (eraserItem.strokeWidth / ch) * pageHeight;
   474→
   475→          for (let i = 0; i < eraserItem.points.length; i++) {
   476→            const p = eraserItem.points[i];
   477→
   478→            const pdfX = (p.x / cw) * pageWidth;
   479→            const pdfY = pageHeight - (p.y / ch) * pageHeight;
   480→
   481→            if (i === 0 && eraserItem.points.length === 1) {
   482→              const halfSize = pdfStrokeWidth / 2;
   483→              page.drawRectangle({
   484→                x: pdfX - halfSize,
   485→                y: pdfY - halfSize,
   486→                width: pdfStrokeWidth,
   487→                height: pdfStrokeWidth,
   488→                color: pdfColor,
   489→              });
   490→            } else if (i > 0) {
   491→              const prevP = eraserItem.points[i - 1];
   492→              const prevPdfX = (prevP.x / cw) * pageWidth;
   493→              const prevPdfY = pageHeight - (prevP.y / ch) * pageHeight;
   494→
   495→              page.drawLine({
   496→                start: { x: prevPdfX, y: prevPdfY },
   497→                end: { x: pdfX, y: pdfY },
   498→                thickness: pdfStrokeWidth,
   499→                color: pdfColor,
   500→              });
   501→            }
   502→          }
   503→        } catch (err) {
   504→          console.error("Error drawing eraser:", err);
   505→        }
   506→      }
   507→
   508→      // Remove pages that are not in keepSet (deleted or not selected for export)
   509→      // Iterate from the highest index to the lowest to avoid shifting issues.
   510→      const indicesToRemove: number[] = [];
   511→      for (let i = 0; i < pageCount; i++) {
   512→        if (!keepSet.has(i + 1)) indicesToRemove.push(i);
   513→      }
   514→      // Sort descending so removal doesn't shift subsequent indices
   515→      indicesToRemove.sort((a, b) => b - a);
   516→      for (const idx of indicesToRemove) {
   517→        try {
   518→          pdfDoc.removePage(idx);
   519→        } catch (err) {
   520→          console.error(`Error removing page ${idx + 1}:`, err);
   521→        }
   522→      }
   523→
   524→      const modifiedPdfBytes = await pdfDoc.save();
   525→      const blob = new Blob([modifiedPdfBytes], { type: "application/pdf" });
   526→      const url = URL.createObjectURL(blob);
   527→
   528→      const link = document.createElement("a");
   529→      link.href = url;
   530→      link.download = pdfFileName
   531→        ? `modified_${pdfFileName}`
   532→        : "modified_document.pdf";
   533→      document.body.appendChild(link);
   534→      link.click();
   535→      document.body.removeChild(link);
   536→
   537→      setTimeout(() => URL.revokeObjectURL(url), 5000);
   538→
   539→      toast.dismiss(loadingToast);
   540→      const exportedCount = pagesToKeep.length;
   541→      toast.success("PDF сохранён!", {
   542→        description: `${exportedCount} стр. · ${stamps.length} печатей · ${texts.length} текстов · ${erasers.length} мазков`,
   543→      });
   544→    } catch (error) {
   545→      toast.dismiss(loadingToast);
   546→      console.error("Error downloading PDF:", error);
   547→      toast.error("Ошибка при сохранении PDF", {
   548→        description: String(error),
   549→      });
   550→    } finally {
   551→      setIsDownloading(false);
   552→    }
   553→  }, [
   554→    pdfArrayBuffer,
   555→    stamps,
   556→    texts,
   557→    erasers,
   558→    pdfFileName,
   559→    isDownloading,
   560→  ]);

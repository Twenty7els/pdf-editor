   361→  // Render current page with zoom
   362→  // zoomLevel = 1.0 means 100% (real PDF size, 1pt = 1px on screen)
   363→  const renderPage = useCallback(async () => {
   364→    if (!pdfDoc || !canvasRef.current || renderingRef.current) return;
   365→    renderingRef.current = true;
   366→    try {
   367→      const pdf = pdfDoc as import("pdfjs-dist").PDFDocumentProxy;
   368→      const page = await pdf.getPage(currentPage);
   369→      const container = containerRef.current;
   370→      if (!container) {
   371→        renderingRef.current = false;
   372→        return;
   373→      }
   374→      // Real PDF scale: 1.0 = 100% (1pt → 1px). No fit-to-container scaling.
   375→      const scale = zoomLevel;
   376→      const rotation = pageRotations[currentPage] || 0;
   377→      const scaledViewport = page.getViewport({ scale, rotation });
   378→      const canvas = canvasRef.current;
   379→      const context = canvas.getContext("2d");
   380→      if (!context) {
   381→        renderingRef.current = false;
   382→        return;
   383→      }
   384→      // Render at higher resolution for crisp text, especially on scanned PDFs.
   385→      // Use max(devicePixelRatio, 2) so even non-retina screens get sharp rendering.
   386→      const outputScale = Math.max(window.devicePixelRatio || 1, 2);
   387→      canvas.width = Math.floor(scaledViewport.width * outputScale);
   388→      canvas.height = Math.floor(scaledViewport.height * outputScale);
   389→      canvas.style.width = Math.floor(scaledViewport.width) + "px";
   390→      canvas.style.height = Math.floor(scaledViewport.height) + "px";
   391→      const transform = [outputScale, 0, 0, outputScale, 0, 0];
   392→      setCanvasSize({
   393→        width: Math.floor(scaledViewport.width),
   394→        height: Math.floor(scaledViewport.height),
   395→      });
   396→      await page.render({
   397→        canvasContext: context,
   398→        viewport: scaledViewport,
   399→        transform,
   400→      }).promise;
   401→    } catch (err) {
   402→      console.error("Error rendering page:", err);
   403→    } finally {
   404→      renderingRef.current = false;
   405→    }
   406→  }, [pdfDoc, currentPage, zoomLevel, pageRotations]);

   581→  }, [

   604→        : canvasSize.width;

   615→        offsetY: e.nativeEvent.offsetY,

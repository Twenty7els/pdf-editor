    34→    34→function hexToRgb(hex: string): { r: number; g: number; b: number } | null {
    35→    35→  const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
    36→    36→  return result
    37→    37→    ? {
    38→    38→        r: parseInt(result[1], 16) / 255,
    39→    39→        g: parseInt(result[2], 16) / 255,
    40→    40→        b: parseInt(result[3], 16) / 255,
    41→    41→      }
    42→    42→    : null;
    43→    43→}

   245→   245→      }

   247→   247→

   258→   258→      let unicodeFonts: {
   259→   259→        regular: unknown;
   260→   260→        bold: unknown;
   261→   261→        italic: unknown;
   262→   262→        boldItalic: unknown;
   263→   263→      } | null = null;
   264→   264→      const getUnicodeFonts = async () => {
   265→   265→        if (unicodeFonts) return unicodeFonts;
   266→   266→        const {
   267→   267→          NOTO_SANS_REGULAR,
   268→   268→          NOTO_SANS_BOLD,
   269→   269→          NOTO_SANS_ITALIC,
   270→   270→          NOTO_SANS_BOLD_ITALIC,
   271→   271→        } = await import("@/lib/font-base64");
   272→   272→        const decode = (b64: string) =>
   273→   273→          new Uint8Array(atob(b64).split("").map((c) => c.charCodeAt(0)));
   274→   274→        unicodeFonts = {
   275→   275→          regular: await pdfDoc.embedFont(decode(NOTO_SANS_REGULAR)),
   276→   276→          bold: await pdfDoc.embedFont(decode(NOTO_SANS_BOLD)),
   277→   277→          italic: await pdfDoc.embedFont(decode(NOTO_SANS_ITALIC)),
   278→   278→          boldItalic: await pdfDoc.embedFont(decode(NOTO_SANS_BOLD_ITALIC)),
   279→   279→        };
   280→   280→        return unicodeFonts;
   281→   281→      };
   282→   282→
   283→   283→      // Stamps — skip if page not in keepSet
   284→   284→      for (const stamp of stamps.filter(
   285→   285→        (s) => !s.hidden && keepSet.has(s.page)
   286→   286→      )) {
   287→   287→        try {
   288→   288→          const page = pdfDoc.getPage(stamp.page - 1);
   289→   289→          const { width: pageWidth, height: pageHeight } = page.getSize();
   290→   290→
   291→   291→          const cw = stamp.canvasWidth || 800;
   292→   292→          const ch = stamp.canvasHeight || 1100;
   293→   293→
   294→   294→          const pdfX = (stamp.x / cw) * pageWidth;
   295→   295→          const pdfY = pageHeight - ((stamp.y + stamp.height) / ch) * pageHeight;
   296→   296→          const pdfWidth = (stamp.width / cw) * pageWidth;
   297→   297→          const pdfHeight = (stamp.height / ch) * pageHeight;
   298→   298→
   299→   299→          // Compensate rotation: pdf-lib rotates around (x, y) = bottom-left corner,
   300→   300→          // but canvas rotates around center. Adjust (x, y) so center stays in place.
   301→   301→          const rad = (stamp.rotation * Math.PI) / 180;
   302→   302→          const cx = pdfX + pdfWidth / 2;
   303→   303→          const cy = pdfY + pdfHeight / 2;
   304→   304→          const adjX = cx - (pdfWidth / 2) * Math.cos(rad) + (pdfHeight / 2) * Math.sin(rad);
   305→   305→          const adjY = cy - (pdfWidth / 2) * Math.sin(rad) - (pdfHeight / 2) * Math.cos(rad);
   306→   306→
   307→   307→          const response = await fetch(stamp.src);
   308→   308→          const imageArrayBuffer = await response.arrayBuffer();
   309→   309→          const imageBytes = new Uint8Array(imageArrayBuffer);
   310→   310→
   311→   311→          let image;
   312→   312→          try {
   313→   313→            image = await pdfDoc.embedPng(imageBytes);
   314→   314→          } catch {
   315→   315→            try {
   316→   316→              image = await pdfDoc.embedJpg(imageBytes);
   317→   317→            } catch {
   318→   318→              console.error("Could not embed stamp image, skipping");
   319→   319→              continue;
   320→   320→            }
   321→   321→          }
   322→   322→
   323→   323→          page.drawImage(image, {
   324→   324→            x: adjX,
   325→   325→            y: adjY,
   326→   326→            width: pdfWidth,
   327→   327→            height: pdfHeight,
   328→   328→            rotate: degrees(stamp.rotation),
   329→   329→            opacity: stamp.opacity,
   330→   330→          });
   331→   331→        } catch (err) {
   332→   332→          console.error("Error embedding stamp:", err);
   333→   333→        }
   334→   334→      }
   335→   335→
   336→   336→      // Texts — skip if page not in keepSet
   337→   337→      for (const textItem of texts.filter(
   338→   338→        (t) => !t.hidden && keepSet.has(t.page)
   339→   339→      )) {
   340→   340→        if (!textItem.text.trim()) continue;
   341→   341→
   342→   342→        try {
   343→   343→          const page = pdfDoc.getPage(textItem.page - 1);
   344→   344→          const { width: pageWidth, height: pageHeight } = page.getSize();
   345→   345→
   346→   346→          const cw = textItem.canvasWidth || 800;
   347→   347→          const ch = textItem.canvasHeight || 1100;
   348→   348→
   349→   349→          const scaledFontSize = (textItem.fontSize / ch) * pageHeight;
   350→   350→          // letterSpacing in canvas px → PDF points (text space)
   351→   351→          const scaledLetterSpacing =
   352→   352→            (textItem.letterSpacing / ch) * pageHeight;
   353→   353→
   354→   354→          let font: import("pdf-lib").PDFFont;
   355→   355→          const uFonts = await getUnicodeFonts();
   356→   356→          if (textItem.bold && textItem.italic) font = uFonts.boldItalic as import("pdf-lib").PDFFont;
   357→   357→          else if (textItem.bold) font = uFonts.bold as import("pdf-lib").PDFFont;
   358→   358→          else if (textItem.italic) font = uFonts.italic as import("pdf-lib").PDFFont;
   359→   359→          else font = uFonts.regular as import("pdf-lib").PDFFont;
   360→   360→
   361→   361→          const color = hexToRgb(textItem.color);
   362→   362→          const pdfColor = color
   363→   363→            ? rgb(color.r, color.g, color.b)
   364→   364→            : rgb(0, 0, 0);
   365→   365→
   366→   366→          // Multiline support
   367→   367→          const lines = textItem.text.split("\n");
   368→   368→
   369→   369→          // Font ascent — distance from baseline to top of glyph.
   370→   370→          // CSS positions text by top of line-box; glyph is centered within line-height.
   371→   371→          // PDF positions text by baseline. So: baseline = top + leading/2 + ascent.
   372→   372→          const fontAscent = font.heightAtSize(scaledFontSize, {
   373→   373→            descender: false,
   374→   374→          });
   375→   375→          // Line height = 1.2 * fontSize (matches canvas CSS lineHeight: 1.2)
   376→   376→          const scaledLineHeight = scaledFontSize * 1.2;
   377→   377→          // Leading is distributed equally above and below the glyph in CSS
   378→   378→          const leadingHalf = (scaledLineHeight - scaledFontSize) / 2;
   379→   379→
   380→   380→          for (let li = 0; li < lines.length; li++) {
   381→   381→            const line = lines[li];
   382→   382→            if (!line) {
   383→   383→              continue;
   384→   384→            }
   385→   385→
   386→   386→            // Canvas: top of line-box = textItem.y + li * (fontSize * 1.2)
   387→   387→            // Glyph top = top of line-box + leading/2
   388→   388→            // PDF baseline = glyph top + ascent
   389→   389→            const lineCanvasTop = textItem.y + li * textItem.fontSize * 1.2;
   390→   390→            const lineBaselinePdfY =
   391→   391→              pageHeight -
   392→   392→              ((lineCanvasTop + leadingHalf + fontAscent) / ch) * pageHeight;
   393→   393→
   394→   394→            // Measure line width for alignment
   395→   395→            const lineWidth = font.widthOfTextAtSize(line, scaledFontSize);
   396→   396→
   397→   397→            // Compute X based on alignment
   398→   398→            let linePdfX: number;
   399→   399→            const basePdfX = (textItem.x / cw) * pageWidth;
   400→   400→            if (textItem.align === "center") {
   401→   401→              linePdfX = basePdfX - lineWidth / 2;
   402→   402→            } else if (textItem.align === "right") {
   403→   403→              linePdfX = basePdfX - lineWidth;
   404→   404→            } else {
   405→   405→              linePdfX = basePdfX;
   406→   406→            }
   407→   407→
   408→   408→            // Compensate rotation for text (pdf-lib rotates around baseline-left,
   409→   409→            // canvas rotates around center). Adjust (x, y) so center stays in place.
   410→   410→            const textRad = (textItem.rotation * Math.PI) / 180;
   411→   411→            const lineCenterX = linePdfX + lineWidth / 2;
   412→   412→            const lineCenterY = lineBaselinePdfY + scaledFontSize / 2;
   413→   413→            const textAdjX =
   414→   414→              lineCenterX -
   415→   415→              (lineWidth / 2) * Math.cos(textRad) +
   416→   416→              (scaledFontSize / 2) * Math.sin(textRad);
   417→   417→            const textAdjY =
   418→   418→              lineCenterY -
   419→   419→              (lineWidth / 2) * Math.sin(textRad) -
   420→   420→              (scaledFontSize / 2) * Math.cos(textRad);
   421→   421→
   422→   422→            page.drawText(line, {
   423→   423→              x: textAdjX,
   424→   424→              y: textAdjY,
   425→   425→              size: scaledFontSize,
   426→   426→              font,
   427→   427→              color: pdfColor,
   428→   428→              rotate: degrees(textItem.rotation),
   429→   429→              // charSpacing is in text space units (1/1000 of font size)
   430→   430→              charSpacing: (scaledLetterSpacing / scaledFontSize) * 1000,
   431→   431→            });
   432→   432→
   433→   433→            // Underline: draw a thin rectangle below the baseline
   434→   434→            if (textItem.underline) {
   435→   435→              const underlineThickness = Math.max(
   436→   436→                0.5,
   437→   437→                scaledFontSize / 18
   438→   438→              );
   439→   439→              const underlineY = textAdjY - scaledFontSize * 0.12;
   440→   440→              page.drawRectangle({
   441→   441→                x: textAdjX,
   442→   442→                y: underlineY,
   443→   443→                width: lineWidth,
   444→   444→                height: underlineThickness,
   445→   445→                color: pdfColor,
   446→   446→                rotate: degrees(textItem.rotation),
   447→   447→              });
   448→   448→            }
   449→   449→          }
   450→   450→        } catch (err) {
   451→   451→          console.error("Error drawing text:", err);
   452→   452→        }
   453→   453→      }
   454→   454→
   455→   455→      // Erasers — skip if page not in keepSet
   456→   456→      for (const eraserItem of erasers.filter(
   457→   457→        (e) => !e.hidden && keepSet.has(e.page)
   458→   458→      )) {
   459→   459→        try {
   460→   460→          if (eraserItem.points.length === 0) continue;
   461→   461→
   462→   462→          const page = pdfDoc.getPage(eraserItem.page - 1);
   463→   463→          const { width: pageWidth, height: pageHeight } = page.getSize();
   464→   464→
   465→   465→          const cw = eraserItem.canvasWidth || 800;
   466→   466→          const ch = eraserItem.canvasHeight || 1100;
   467→   467→
   468→   468→          const eraserColor = hexToRgb(eraserItem.color);
   469→   469→          const pdfColor = eraserColor
   470→   470→            ? rgb(eraserColor.r, eraserColor.g, eraserColor.b)
   471→   471→            : rgb(1, 1, 1);
   472→   472→
   473→   473→          const pdfStrokeWidth = (eraserItem.strokeWidth / ch) * pageHeight;
   474→   474→
   475→   475→          for (let i = 0; i < eraserItem.points.length; i++) {
   476→   476→            const p = eraserItem.points[i];
   477→   477→
   478→   478→            const pdfX = (p.x / cw) * pageWidth;
   479→   479→            const pdfY = pageHeight - (p.y / ch) * pageHeight;
   480→   480→
   481→   481→            if (i === 0 && eraserItem.points.length === 1) {
   482→   482→              const halfSize = pdfStrokeWidth / 2;
   483→   483→              page.drawRectangle({
   484→   484→                x: pdfX - halfSize,
   485→   485→                y: pdfY - halfSize,
   486→   486→                width: pdfStrokeWidth,
   487→   487→                height: pdfStrokeWidth,
   488→   488→                color: pdfColor,
   489→   489→              });
   490→   490→            } else if (i > 0) {
   491→   491→              const prevP = eraserItem.points[i - 1];
   492→   492→              const prevPdfX = (prevP.x / cw) * pageWidth;
   493→   493→              const prevPdfY = pageHeight - (prevP.y / ch) * pageHeight;
   494→   494→
   495→   495→              page.drawLine({
   496→   496→                start: { x: prevPdfX, y: prevPdfY },
   497→   497→                end: { x: pdfX, y: pdfY },
   498→   498→                thickness: pdfStrokeWidth,
   499→   499→                color: pdfColor,
   500→   500→              });
   501→   501→            }
   502→   502→          }
   503→   503→        } catch (err) {
   504→   504→          console.error("Error drawing eraser:", err);
   505→   505→        }
   506→   506→      }
   507→   507→
   508→   508→      // Remove pages that are not in keepSet (deleted or not selected for export)
   509→   509→      // Iterate from the highest index to the lowest to avoid shifting issues.
   510→   510→      const indicesToRemove: number[] = [];
   511→   511→      for (let i = 0; i < pageCount; i++) {
   512→   512→        if (!keepSet.has(i + 1)) indicesToRemove.push(i);
   513→   513→      }
   514→   514→      // Sort descending so removal doesn't shift subsequent indices
   515→   515→      indicesToRemove.sort((a, b) => b - a);
   516→   516→      for (const idx of indicesToRemove) {

   518→   518→          pdfDoc.removePage(idx);
   519→   519→        } catch (err) {

   521→   521→        }
   522→   522→      }

   527→   527→
   528→   528→      const link = document.createElement("a");
   529→   529→      link.href = url;
   530→   530→      link.download = pdfFileName
   531→   531→        ? `modified_${pdfFileName}`
   532→   532→        : "modified_document.pdf";
   533→   533→      document.body.appendChild(link);
   534→   534→      link.click();
   535→   535→      document.body.removeChild(link);
   536→   536→
   537→   537→      setTimeout(() => URL.revokeObjectURL(url), 5000);
   538→   538→
   539→   539→      toast.dismiss(loadingToast);
   540→   540→      const exportedCount = pagesToKeep.length;
   541→   541→      toast.success("PDF сохранён!", {
   542→   542→        description: `${exportedCount} стр. · ${stamps.length} печатей · ${texts.length} текстов · ${erasers.length} мазков`,
   543→   543→      });
   544→   544→    } catch (error) {
   545→   545→      toast.dismiss(loadingToast);
   546→   546→      console.error("Error downloading PDF:", error);
   547→   547→      toast.error("Ошибка при сохранении PDF", {
   548→   548→        description: String(error),
   549→   549→      });
   550→   550→    } finally {
   551→   551→      setIsDownloading(false);
   552→   552→    }
   553→   553→  }, [
   554→   554→    pdfArrayBuffer,
   555→   555→    stamps,
   556→   556→    texts,
   557→   557→    erasers,
   558→   558→    pdfFileName,
   559→   559→    isDownloading,
   560→   560→  ]);

   573→   573→    />

   575→   575→
   576→   576→  if (!authenticated) {
   577→   577→    return <LoginScreen onLogin={handleLogin} />;
   578→   578→  }

   580→   580→  return (
   581→   581→    <div className="min-h-screen flex flex-col bg-background">
   582→   582→      {/* Header */}
   583→   583→      <header className="border-b border-border/40 glass-strong px-4 py-2.5 flex items-center justify-between shrink-0 z-30">
   584→   584→        <div className="flex items-center gap-3">
   585→   585→          {pdfFile && (
   586→   586→            <Sheet open={sidebarOpen} onOpenChange={setSidebarOpen}>
   587→   587→              <SheetTrigger asChild>
   588→   588→                <Button
   589→   589→                  variant="ghost"
   590→   590→                  size="icon"
   591→   591→                  className="md:hidden h-9 w-9 rounded-xl"
   592→   592→                >
   593→   593→                  <Menu className="h-5 w-5" />
   594→   594→                </Button>
   595→   595→              </SheetTrigger>
   596→   596→              <SheetContent side="left" className="p-0 w-72 overflow-y-auto">
   597→   597→                <SheetTitle className="sr-only">Панель инструментов</SheetTitle>
   598→   598→                {toolbarContent}
   599→   599→                <div className="h-px bg-gradient-to-r from-transparent via-border/40 to-transparent mx-4" />

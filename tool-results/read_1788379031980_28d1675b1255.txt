    33→    33→function hexToRgb(hex: string): { r: number; g: number; b: number } | null {
    34→    34→  const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
    35→    35→  return result
    36→    36→    ? {
    37→    37→        r: parseInt(result[1], 16) / 255,
    38→    38→        g: parseInt(result[2], 16) / 255,
    39→    39→        b: parseInt(result[3], 16) / 255,
    40→    40→      }
    41→    41→    : null;
    42→    42→}

    44→    44→/* ============================================================
    45→    45→   Export geometry helpers
    46→    46→   ------------------------------------------------------------
    47→    47→   Items are stored in "view space" — the coordinate system of
    48→    48→   the rotated page as shown on screen (top-left origin, y down).
    49→    49→   The exported PDF page keeps an unrotated mediabox and gets a
    50→    50→   /Rotate entry, so every point and angle must be transformed.
    51→    51→   ============================================================ */
    52→    52→
    53→    53→interface PageSize {
    54→    54→  width: number;
    55→    55→  height: number;
    56→    56→}
    57→    57→
    58→    58→/** Display (view) dimensions in points for a page rotated T degrees CW. */
    59→    59→function getViewDims(T: number, size: PageSize): PageSize {
    60→    60→  const swap = T === 90 || T === 270;
    61→    61→  return swap
    62→    62→    ? { width: size.height, height: size.width }
    63→    63→    : { width: size.width, height: size.height };
    64→    64→}

   291→   291→          const existing = page.getRotation().angle || 0;

   294→   294→        }

   297→   297→      let unicodeFonts: {
   298→   298→        regular: unknown;
   299→   299→        bold: unknown;
   300→   300→        italic: unknown;
   301→   301→        boldItalic: unknown;
   302→   302→      } | null = null;

   304→   304→        if (unicodeFonts) return unicodeFonts;

   306→   306→          NOTO_SANS_REGULAR,
   307→   307→          NOTO_SANS_BOLD,
   308→   308→          NOTO_SANS_ITALIC,
   309→   309→          NOTO_SANS_BOLD_ITALIC,
   310→   310→        } = await import("@/lib/font-base64");
   311→   311→        const decode = (b64: string) =>
   312→   312→          new Uint8Array(atob(b64).split("").map((c) => c.charCodeAt(0)));
   313→   313→        unicodeFonts = {
   314→   314→          regular: await pdfDoc.embedFont(decode(NOTO_SANS_REGULAR)),
   315→   315→          bold: await pdfDoc.embedFont(decode(NOTO_SANS_BOLD)),
   316→   316→          italic: await pdfDoc.embedFont(decode(NOTO_SANS_ITALIC)),
   317→   317→          boldItalic: await pdfDoc.embedFont(decode(NOTO_SANS_BOLD_ITALIC)),
   318→   318→        };

   321→   321→
   322→   322→      /**
   323→   323→       * Convert a point from view space (top-left origin, y down, points)
   324→   324→       * to PDF space (bottom-left origin, y up) for a page with total
   325→   325→       * rotation T. `dims` = unrotated mediabox {width W, height H},
   326→   326→       * `Wv`/`Hv` = rotated view dimensions.
   327→   327→       */
   328→   328→      const viewPointToPdf = (
   329→   329→        vx: number,
   330→   330→        vy: number,
   331→   331→        T: number,
   332→   332→        H: number,
   333→   333→        Wv: number,
   334→   334→        Hv: number
   335→   335→      ): { x: number; y: number } => {
   336→   336→        let px: number;
   337→   337→        let py: number;
   338→   338→        switch (T) {
   339→   339→          case 90:
   340→   340→            px = vy;
   341→   341→            py = Wv - vx;
   342→   342→            break;
   343→   343→          case 180:
   344→   344→            px = Wv - vx;
   345→   345→            py = Hv - vy;
   346→   346→            break;
   347→   347→          case 270:
   348→   348→            px = Hv - vy;
   349→   349→            py = vx;
   350→   350→            break;
   351→   351→          default:
   352→   352→            px = vx;
   353→   353→            py = vy;
   354→   354→        }
   355→   355→        return { x: px, y: H - py };
   356→   356→      };
   357→   357→
   358→   358→      // Stamps — skip if page not in keepSet

   362→   362→          const { width: pageWidth, height: pageHeight } = page.getSize();
   363→   363→
   364→   364→          const userRotation = pageRotations[stamp.page] || 0;
   365→   365→          const intrinsic = page.getRotation().angle || 0;
   366→   366→          const T = (((intrinsic + userRotation) % 360) + 360) % 360;
   367→   367→          const { width: Wv, height: Hv } = getViewDims(T, {

   369→   369→            height: pageHeight,

   379→   379→          // Item center in view space → PDF space

   381→   381→          const cvy = ((stamp.y + stamp.height / 2) / ch) * Hv;
   382→   382→          const center = viewPointToPdf(cvx, cvy, T, pageHeight, Wv, Hv);
   383→   383→
   384→   384→          // pdf-lib rotates CCW; canvas rotation is CW → invert,
   385→   385→          // and account for the page's own /Rotate T.
   386→   386→          const phiDeg = T - stamp.rotation;
   387→   387→          const rad = (phiDeg * Math.PI) / 180;
   388→   388→
   389→   389→          // Anchor so the rotated image stays centered on `center`
   390→   390→          const adjX =
   391→   391→            center.x - (pw / 2) * Math.cos(rad) + (ph / 2) * Math.sin(rad);
   392→   392→          const adjY =
   393→   393→            center.y - (pw / 2) * Math.sin(rad) - (ph / 2) * Math.cos(rad);
   394→   394→
   395→   395→          const response = await fetch(stamp.src);
   396→   396→          const imageArrayBuffer = await response.arrayBuffer();
   397→   397→          const imageBytes = new Uint8Array(imageArrayBuffer);
   398→   398→
   399→   399→          let image;
   400→   400→          try {
   401→   401→            image = await pdfDoc.embedPng(imageBytes);
   402→   402→          } catch {
   403→   403→            try {
   404→   404→              image = await pdfDoc.embedJpg(imageBytes);
   405→   405→            } catch {
   406→   406→              console.error("Could not embed stamp image, skipping");
   407→   407→              continue;
   408→   408→            }
   409→   409→          }
   410→   410→
   411→   411→          page.drawImage(image, {
   412→   412→            x: adjX,
   413→   413→            y: adjY,
   414→   414→            width: pw,
   415→   415→            height: ph,
   416→   416→            rotate: degrees(phiDeg),
   417→   417→            opacity: stamp.opacity,
   418→   418→          });
   419→   419→        } catch (err) {
   420→   420→          console.error("Error embedding stamp:", err);
   421→   421→        }
   422→   422→      }
   423→   423→
   424→   424→      // Texts — skip if page not in keepSet
   425→   425→      for (const textItem of exportedTexts) {
   426→   426→        try {
   427→   427→          const page = pdfDoc.getPage(textItem.page - 1);
   428→   428→          const { width: pageWidth, height: pageHeight } = page.getSize();
   429→   429→
   430→   430→          const userRotation = pageRotations[textItem.page] || 0;
   431→   431→          const intrinsic = page.getRotation().angle || 0;
   432→   432→          const T = (((intrinsic + userRotation) % 360) + 360) % 360;
   433→   433→          const { width: Wv, height: Hv } = getViewDims(T, {
   434→   434→            width: pageWidth,
   435→   435→            height: pageHeight,
   436→   436→          });
   437→   437→
   438→   438→          const cw = textItem.canvasWidth || Wv;
   439→   439→          const ch = textItem.canvasHeight || Hv;
   440→   440→
   441→   441→          const scaledFontSize = (textItem.fontSize / ch) * Hv;
   442→   442→          const scaledLetterSpacing = (textItem.letterSpacing / cw) * Wv;
   443→   443→
   444→   444→          let font: import("pdf-lib").PDFFont;
   445→   445→          const uFonts = await getUnicodeFonts();
   446→   446→          if (textItem.bold && textItem.italic)
   447→   447→            font = uFonts.boldItalic as import("pdf-lib").PDFFont;
   448→   448→          else if (textItem.bold)
   449→   449→            font = uFonts.bold as import("pdf-lib").PDFFont;
   450→   450→          else if (textItem.italic)
   451→   451→            font = uFonts.italic as import("pdf-lib").PDFFont;
   452→   452→          else font = uFonts.regular as import("pdf-lib").PDFFont;
   453→   453→
   454→   454→          const color = hexToRgb(textItem.color);
   455→   455→          const pdfColor = color
   456→   456→            ? rgb(color.r, color.g, color.b)
   457→   457→            : rgb(0, 0, 0);
   458→   458→
   459→   459→          // Multiline support
   460→   460→          const lines = textItem.text.split("\n");
   461→   461→
   462→   462→          // Font ascent — distance from baseline to top of glyph.
   463→   463→          // CSS positions text by top of line-box; glyph is centered within line-height.
   464→   464→          // PDF positions text by baseline. So: baseline = top + leading/2 + ascent.
   465→   465→          const fontAscent = font.heightAtSize(scaledFontSize, {
   466→   466→            descender: false,
   467→   467→          });
   468→   468→          // Line height = 1.2 * fontSize (matches canvas CSS lineHeight: 1.2)
   469→   469→          const scaledLineHeight = scaledFontSize * 1.2;
   470→   470→          const leadingHalf = (scaledLineHeight - scaledFontSize) / 2;
   471→   471→
   472→   472→          // pdf-lib CCW angle that matches the canvas CW angle after page rotation
   473→   473→          const phiDeg = T - textItem.rotation;
   474→   474→          const phiRad = (phiDeg * Math.PI) / 180;
   475→   475→
   476→   476→          for (let li = 0; li < lines.length; li++) {
   477→   477→            const line = lines[li];
   478→   478→            if (!line) {
   479→   479→              continue;
   480→   480→            }
   481→   481→
   482→   482→            // Canvas: top of line-box = textItem.y + li * (fontSize * 1.2)
   483→   483→            const lineCanvasTop = textItem.y + li * textItem.fontSize * 1.2;
   484→   484→            const baselineViewY =
   485→   485→              (lineCanvasTop / ch) * Hv + leadingHalf + fontAscent;
   486→   486→
   487→   487→            // pdf-lib has no charSpacing option — emulate letter spacing by
   488→   488→            // drawing glyph runs per character when it is non-zero.
   489→   489→            const hasSpacing = Math.abs(scaledLetterSpacing) > 0.001;
   490→   490→            const chars = hasSpacing ? Array.from(line) : null;
   491→   491→            const charWidths = chars
   492→   492→              ? chars.map((c) => font.widthOfTextAtSize(c, scaledFontSize))
   493→   493→              : null;
   494→   494→            const totalSpacing = charWidths
   495→   495→              ? scaledLetterSpacing * Math.max(0, chars!.length - 1)
   496→   496→              : 0;
   497→   497→            const naturalWidth = font.widthOfTextAtSize(line, scaledFontSize);
   498→   498→            const lineWidth = charWidths
   499→   499→              ? charWidths.reduce((a, b) => a + b, 0) + totalSpacing
   500→   500→              : naturalWidth;
   501→   501→
   502→   502→            // Compute baseline X in view space based on alignment
   503→   503→            const baseViewX = (textItem.x / cw) * Wv;
   504→   504→            const baselineViewX =
   505→   505→              textItem.align === "center"
   506→   506→                ? baseViewX - lineWidth / 2
   507→   507→                : textItem.align === "right"
   508→   508→                ? baseViewX - lineWidth
   509→   509→                : baseViewX;
   510→   510→
   511→   511→            // Baseline start point → PDF space
   512→   512→            const baselinePdf = viewPointToPdf(
   513→   513→              baselineViewX,
   514→   514→              baselineViewY,
   515→   515→              T,
   516→   516→              pageHeight,
   517→   517→              Wv,
   518→   518→              Hv
   519→   519→            );
   520→   520→
   521→   521→            // Treat the line box as (lineWidth × fontSize) centered on
   522→   522→            // baseline + fontSize/2; anchor so rotation keeps center fixed.
   523→   523→            const lineCenterX = baselinePdf.x + lineWidth / 2;
   524→   524→            const lineCenterY = baselinePdf.y + scaledFontSize / 2;
   525→   525→            const textAdjX =
   526→   526→              lineCenterX -
   527→   527→              (lineWidth / 2) * Math.cos(phiRad) +
   528→   528→              (scaledFontSize / 2) * Math.sin(phiRad);
   529→   529→            const textAdjY =
   530→   530→              lineCenterY -
   531→   531→              (lineWidth / 2) * Math.sin(phiRad) -
   532→   532→              (scaledFontSize / 2) * Math.cos(phiRad);
   533→   533→
   534→   534→            const drawOpts = {
   535→   535→              x: textAdjX,
   536→   536→              y: textAdjY,
   537→   537→              size: scaledFontSize,
   538→   538→              font,
   539→   539→              color: pdfColor,
   540→   540→              rotate: degrees(phiDeg),
   541→   541→            };
   542→   542→
   543→   543→            if (charWidths && chars) {
   544→   544→              // Letter-spaced: place each character along the rotated baseline
   545→   545→              let cursorX = textAdjX;
   546→   546→              let cursorY = textAdjY;
   547→   547→              for (let ci = 0; ci < chars.length; ci++) {
   548→   548→                if (chars[ci] !== " ") {
   549→   549→                  page.drawText(chars[ci], { ...drawOpts, x: cursorX, y: cursorY });
   550→   550→                }
   551→   551→                const adv = charWidths[ci] + scaledLetterSpacing;
   552→   552→                cursorX += adv * Math.cos(phiRad);
   553→   553→                cursorY += adv * Math.sin(phiRad);
   554→   554→              }
   555→   555→            } else {
   556→   556→              page.drawText(line, drawOpts);
   557→   557→            }
   558→   558→
   559→   559→            // Underline: thin rectangle below the baseline, offset
   560→   560→            // perpendicular to the text direction so it survives rotation.
   561→   561→            if (textItem.underline) {
   562→   562→              const underlineThickness = Math.max(0.5, scaledFontSize / 18);
   563→   563→              const off = scaledFontSize * 0.12;
   564→   564→              page.drawRectangle({
   565→   565→                x: textAdjX + off * Math.sin(phiRad),
   566→   566→                y: textAdjY - off * Math.cos(phiRad),
   567→   567→                width: lineWidth,
   568→   568→                height: underlineThickness,
   569→   569→                color: pdfColor,
   570→   570→                rotate: degrees(phiDeg),
   571→   571→              });
   572→   572→            }
   573→   573→          }
   574→   574→        } catch (err) {
   575→   575→          console.error("Error drawing text:", err);
   576→   576→        }
   577→   577→      }
   578→   578→
   579→   579→      // Erasers — skip if page not in keepSet
   580→   580→      for (const eraserItem of exportedErasers) {
   581→   581→        try {
   582→   582→          const page = pdfDoc.getPage(eraserItem.page - 1);
   583→   583→          const { width: pageWidth, height: pageHeight } = page.getSize();
   584→   584→
   585→   585→          const userRotation = pageRotations[eraserItem.page] || 0;
   586→   586→          const intrinsic = page.getRotation().angle || 0;
   587→   587→          const T = (((intrinsic + userRotation) % 360) + 360) % 360;
   588→   588→          const { width: Wv, height: Hv } = getViewDims(T, {
   589→   589→            width: pageWidth,
   590→   590→            height: pageHeight,
   591→   591→          });
   592→   592→
   593→   593→          const cw = eraserItem.canvasWidth || Wv;
   594→   594→          const ch = eraserItem.canvasHeight || Hv;
   595→   595→
   596→   596→          const eraserColor = hexToRgb(eraserItem.color);
   597→   597→          const pdfColor = eraserColor
   598→   598→            ? rgb(eraserColor.r, eraserColor.g, eraserColor.b)
   599→   599→            : rgb(1, 1, 1);
   600→   600→
   601→   601→          const pdfStrokeWidth = (eraserItem.strokeWidth / ch) * Hv;
   602→   602→
   603→   603→          const pdfPoints = eraserItem.points.map((p) => {
   604→   604→            const vx = (p.x / cw) * Wv;
   605→   605→            const vy = (p.y / ch) * Hv;
   606→   606→            return viewPointToPdf(vx, vy, T, pageHeight, Wv, Hv);
   607→   607→          });
   608→   608→
   609→   609→          for (let i = 0; i < pdfPoints.length; i++) {
   610→   610→            const p = pdfPoints[i];
   611→   611→
   612→   612→            if (i === 0 && pdfPoints.length === 1) {
   613→   613→              const halfSize = pdfStrokeWidth / 2;
   614→   614→              page.drawRectangle({
   615→   615→                x: p.x - halfSize,
   616→   616→                y: p.y - halfSize,
   617→   617→                width: pdfStrokeWidth,
   618→   618→                height: pdfStrokeWidth,
   619→   619→                color: pdfColor,
   620→   620→              });
   621→   621→            } else if (i > 0) {
   622→   622→              const prevP = pdfPoints[i - 1];
   623→   623→
   624→   624→              page.drawLine({
   625→   625→                start: { x: prevP.x, y: prevP.y },
   626→   626→                end: { x: p.x, y: p.y },
   627→   627→                thickness: pdfStrokeWidth,
   628→   628→                color: pdfColor,
   629→   629→              });
   630→   630→            }
   631→   631→          }
   632→   632→        } catch (err) {
   633→   633→          console.error("Error drawing eraser:", err);
   634→   634→        }
   635→   635→      }
   636→   636→
   637→   637→      // Remove pages that are not in keepSet (deleted or not selected for export)
   638→   638→      // Iterate from the highest index to the lowest to avoid shifting issues.
   639→   639→      const indicesToRemove: number[] = [];
   640→   640→      for (let i = 0; i < pageCount; i++) {
   641→   641→        if (!keepSet.has(i + 1)) indicesToRemove.push(i);
   642→   642→      }
   643→   643→      // Sort descending so removal doesn't shift subsequent indices
   644→   644→      indicesToRemove.sort((a, b) => b - a);
   645→   645→      for (const idx of indicesToRemove) {
   646→   646→        try {
   647→   647→          pdfDoc.removePage(idx);
   648→   648→        } catch (err) {
   649→   649→          console.error(`Error removing page ${idx + 1}:`, err);
   650→   650→        }
   651→   651→      }
   652→   652→
   653→   653→      const modifiedPdfBytes = await pdfDoc.save();
   654→   654→      const blob = new Blob([new Uint8Array(modifiedPdfBytes)], {
   655→   655→        type: "application/pdf",
   656→   656→      });
   657→   657→      const url = URL.createObjectURL(blob);
   658→   658→
   659→   659→      const link = document.createElement("a");
   660→   660→      link.href = url;
   661→   661→      link.download = pdfFileName
   662→   662→        ? `modified_${pdfFileName}`
   663→   663→        : "modified_document.pdf";
   664→   664→      document.body.appendChild(link);
   665→   665→      link.click();
   666→   666→      document.body.removeChild(link);
   667→   667→
   668→   668→      setTimeout(() => URL.revokeObjectURL(url), 5000);
   669→   669→
   670→   670→      toast.dismiss(loadingToast);
   671→   671→      toast.success("PDF сохранён!", {
   672→   672→        description: `${pagesToKeep.length} стр. · ${exportedStamps.length} печатей · ${exportedTexts.length} текстов · ${exportedErasers.length} мазков`,
   673→   673→      });
   674→   674→    } catch (error) {
   675→   675→      toast.dismiss(loadingToast);
   676→   676→      console.error("Error downloading PDF:", error);
   677→   677→      toast.error("Ошибка при сохранении PDF", {
   678→   678→        description: String(error),
   679→   679→      });
   680→   680→    } finally {
   681→   681→      setIsDownloading(false);
   682→   682→    }
   683→   683→  }, [
   684→   684→    pdfArrayBuffer,
   685→   685→    stamps,
   686→   686→    texts,
   687→   687→    erasers,
   688→   688→    pdfFileName,
   689→   689→    isDownloading,
   690→   690→  ]);
   691→   691→
   692→   692→  const toolbarContent = (
   693→   693→    <Toolbar
   694→   694→      onUploadClick={() => {
   695→   695→        handleUploadClick();
   696→   696→        setSidebarOpen(false);
   697→   697→      }}
   698→   698→      onDownloadClick={() => {
   699→   699→        setExportDialogOpen(true);
   700→   700→        setSidebarOpen(false);
   701→   701→      }}
   702→   702→      isDownloading={isDownloading}
   703→   703→    />
   704→   704→  );
   705→   705→
   706→   706→  if (!authenticated) {
   707→   707→    return <LoginScreen onLogin={handleLogin} />;
   708→   708→  }
   709→   709→
   710→   710→  return (
   711→   711→    <div className="min-h-screen flex flex-col bg-background">
   712→   712→      {/* Header */}
   713→   713→      <header className="border-b border-border/70 glass-strong px-4 py-2.5 flex items-center justify-between shrink-0 z-30">
   714→   714→        <div className="flex items-center gap-3">
   715→   715→          {pdfFile && (
   716→   716→            <Sheet open={sidebarOpen} onOpenChange={setSidebarOpen}>
   717→   717→              <SheetTrigger asChild>
   718→   718→                <Button
   719→   719→                  variant="ghost"
   720→   720→                  size="icon"
   721→   721→                  className="md:hidden h-9 w-9 rounded-xl"
   722→   722→                >
   723→   723→                  <Menu className="h-5 w-5" />
   724→   724→                </Button>
   725→   725→              </SheetTrigger>
   726→   726→              <SheetContent side="left" className="p-0 w-72 overflow-y-auto">
   727→   727→                <SheetTitle className="sr-only">Панель инструментов</SheetTitle>
   728→   728→                {toolbarContent}
   729→   729→                <div className="h-px bg-border/60 mx-4" />
   730→   730→                <LayersPanel currentPage={currentPage} totalPages={totalPages} />
   731→   731→              </SheetContent>
   732→   732→            </Sheet>
   733→   733→          )}
   734→   734→
   735→   735→          {/* Logo */}
   736→   736→          <div className="relative h-9 w-9 rounded-xl gradient-bg flex items-center justify-center shadow-soft">
   737→   737→            <FileText
   738→   738→              className="h-5 w-5 text-white"
   739→   739→              strokeWidth={2}
   740→   740→              fill="rgba(255,255,255,0.15)"
   741→   741→            />
   742→   742→          </div>
   743→   743→          <div>
   744→   744→            <h1 className="font-display text-base font-semibold leading-tight tracking-tight">
   745→   745→              PDF{" "}
   746→   746→              <em className="italic font-medium text-primary">Редактор</em>
   747→   747→            </h1>
   748→   748→            <p className="text-[11px] text-muted-foreground hidden sm:block font-medium">
   749→   749→              Печати и текст на документах
   750→   750→            </p>
   751→   751→          </div>
   752→   752→          {pdfFile && (
   753→   753→            <div className="hidden lg:flex items-center gap-1.5 ml-6">

   763→   763→                >

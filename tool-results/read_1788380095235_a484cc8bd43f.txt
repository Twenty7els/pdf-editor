   362→            Math.abs(h - A4_WIDTH) <= TOLERANCE;

   382→  // Render current page with zoom
   383→  // zoomLevel = 1.0 means 100% (real PDF size, 1pt = 1px on screen)
   384→  const renderPage = useCallback(async () => {
   385→    if (!pdfDoc || !canvasRef.current || renderingRef.current) return;
   386→    renderingRef.current = true;
   387→    try {
   388→      const page = await pdfDoc.getPage(currentPage);
   389→      const container = containerRef.current;
   390→      if (!container) {
   391→        renderingRef.current = false;
   392→        return;
   393→      }
   394→      // Real PDF scale: 1.0 = 100% (1pt → 1px). No fit-to-container scaling.
   395→      const scale = zoomLevel;
   396→      // Total rotation = intrinsic page rotation + user rotation.
   397→      // pdf.js treats the `rotation` option as ABSOLUTE, so we must add
   398→      // the page's own /Rotate value — otherwise intrinsically rotated
   399→      // pages render differently here than in the exported PDF.
   400→      const intrinsic = page.rotate || 0;
   401→      const userRotation = pageRotations[currentPage] || 0;
   402→      const rotation =
   403→        ((((intrinsic + userRotation) % 360) + 360) % 360) as number;
   404→      const scaledViewport = page.getViewport({ scale, rotation });
   405→      const canvas = canvasRef.current;
   406→      const context = canvas.getContext("2d");
   407→      if (!context) {
   408→        renderingRef.current = false;
   409→        return;
   410→      }
   411→      // Render at higher resolution for crisp text, especially on scanned PDFs.
   412→      // Use max(devicePixelRatio, 2) so even non-retina screens get sharp rendering.
   413→      const outputScale = Math.max(window.devicePixelRatio || 1, 2);
   414→      canvas.width = Math.floor(scaledViewport.width * outputScale);
   415→      canvas.height = Math.floor(scaledViewport.height * outputScale);
   416→      canvas.style.width = Math.floor(scaledViewport.width) + "px";
   417→      canvas.style.height = Math.floor(scaledViewport.height) + "px";
   418→      const transform = [outputScale, 0, 0, outputScale, 0, 0];
   419→      setCanvasSize({
   420→        width: Math.floor(scaledViewport.width),
   421→        height: Math.floor(scaledViewport.height),
   422→      });
   423→      await page.render({
   424→        canvas,
   425→        canvasContext: context,
   426→        viewport: scaledViewport,
   427→        transform,
   428→      }).promise;
   429→    } catch (err) {
   430→      console.error("Error rendering page:", err);
   431→    } finally {
   432→      renderingRef.current = false;
   433→    }

   601→      setIsEraserDrawing(false);
   602→    };
   603→

   605→    window.addEventListener("mouseup", handleMouseUp);
   606→    return () => {
   607→      window.removeEventListener("mousemove", handleMouseMove);
   608→      window.removeEventListener("mouseup", handleMouseUp);
   609→    };
   610→  }, [

    33→function hexToRgb(hex: string): { r: number; g: number; b: number } | null {
    34→  const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
    35→  return result
    36→    ? {
    37→        r: parseInt(result[1], 16) / 255,
    38→        g: parseInt(result[2], 16) / 255,
    39→        b: parseInt(result[3], 16) / 255,
    40→      }
    41→    : null;
    42→}

    44→/* ============================================================
    45→   Export geometry helpers
    46→   ------------------------------------------------------------
    47→   Items are stored in "view space" — the coordinate system of
    48→   the rotated page as shown on screen (top-left origin, y down).
    49→   The exported PDF page keeps an unrotated mediabox and gets a
    50→   /Rotate entry, so every point and angle must be transformed.
    51→   ============================================================ */
    52→
    53→interface PageSize {
    54→  width: number;
    55→  height: number;
    56→}
    57→
    58→/** Display (view) dimensions in points for a page rotated T degrees CW. */
    59→function getViewDims(T: number, size: PageSize): PageSize {
    60→  const swap = T === 90 || T === 270;
    61→  return swap
    62→    ? { width: size.height, height: size.width }
    63→    : { width: size.width, height: size.height };
    64→}

   189→export default function Home() {
   190→  const {
   191→    pdfFile,
   192→    pdfArrayBuffer,
   193→    stamps,
   194→    texts,
   195→    erasers,
   196→    pdfFileName,
   197→    currentPage,

   199→    setPdfFile,

   201→
   202→  const [authenticated, setAuthenticatedState] = useState(false);
   203→  const fileInputRef = useRef<HTMLInputElement>(null);
   204→  const [sidebarOpen, setSidebarOpen] = useState(false);
   205→  const [isDownloading, setIsDownloading] = useState(false);
   206→  const [exportDialogOpen, setExportDialogOpen] = useState(false);
   207→
   208→  useEffect(() => {

   210→      setAuthenticatedState(true);
   211→    }

   214→  const handleLogin = useCallback(() => {

   221→    usePdfEditorStore.getState().reset();
   222→  }, []);
   223→
   224→  const handleUploadClick = useCallback(() => {
   225→    fileInputRef.current?.click();
   226→  }, []);
   227→
   228→  const handleFileChange = useCallback(
   229→    (e: React.ChangeEvent<HTMLInputElement>) => {
   230→      const file = e.target.files?.[0];
   231→      if (file && file.type === "application/pdf") {
   232→        setPdfFile(file);
   233→        toast.success("PDF загружен", {
   234→          description: file.name,
   235→        });
   236→      } else {
   237→        toast.error("Пожалуйста, выберите PDF файл");
   238→      }
   239→      if (fileInputRef.current) {
   240→        fileInputRef.current.value = "";
   241→      }
   242→    },
   243→    [setPdfFile]
   244→  );
   245→
   246→  const handleDownload = useCallback(async () => {
   247→    if (!pdfArrayBuffer || isDownloading) return;
   248→
   249→    setIsDownloading(true);
   250→    const loadingToast = toast.loading("Подготовка PDF...");
   251→
   252→    try {
   253→      const { PDFDocument, rgb, degrees } = await import("pdf-lib");
   254→
   255→      const pdfDoc = await PDFDocument.load(pdfArrayBuffer);
   256→
   257→      const fontkit = (await import("@pdf-lib/fontkit")).default;
   258→      pdfDoc.registerFontkit(fontkit);
   259→
   260→      // Read document-level state from store
   261→      const state = usePdfEditorStore.getState();
   262→      const { pageRotations, deletedPages, exportPageSelection } = state;
   263→
   264→      // Determine which pages (1-indexed) to keep in the export
   265→      const pageCount = pdfDoc.getPageCount();
   266→      const pagesToKeep: number[] = [];
   267→      for (let p = 1; p <= pageCount; p++) {
   268→        if (deletedPages.includes(p)) continue;
   269→        if (exportPageSelection !== null && !exportPageSelection.includes(p))
   270→          continue;
   271→        pagesToKeep.push(p);
   272→      }
   273→      const keepSet = new Set(pagesToKeep);
   274→
   275→      // Items actually included in the export (visible & on kept pages)
   276→      const exportedStamps = stamps.filter(
   277→        (s) => !s.hidden && keepSet.has(s.page)
   278→      );
   279→      const exportedTexts = texts.filter(
   280→        (t) => !t.hidden && keepSet.has(t.page) && t.text.trim()
   281→      );
   282→      const exportedErasers = erasers.filter(
   283→        (e) => !e.hidden && keepSet.has(e.page) && e.points.length > 0
   284→      );
   285→
   286→      // Apply page rotations to kept pages BEFORE drawing items
   287→      for (const pageNum of pagesToKeep) {
   288→        const userRotation = pageRotations[pageNum] || 0;
   289→        if (userRotation !== 0) {
   290→          const page = pdfDoc.getPage(pageNum - 1);
   291→          const existing = page.getRotation().angle || 0;
   292→          const total = (((existing + userRotation) % 360) + 360) % 360;
   293→          page.setRotation(degrees(total));
   294→        }
   295→      }
   296→
   297→      let unicodeFonts: {
   298→        regular: unknown;
   299→        bold: unknown;
   300→        italic: unknown;
   301→        boldItalic: unknown;
   302→      } | null = null;
   303→      const getUnicodeFonts = async () => {
   304→        if (unicodeFonts) return unicodeFonts;
   305→        const {
   306→          NOTO_SANS_REGULAR,
   307→          NOTO_SANS_BOLD,
   308→          NOTO_SANS_ITALIC,
   309→          NOTO_SANS_BOLD_ITALIC,
   310→        } = await import("@/lib/font-base64");
   311→        const decode = (b64: string) =>
   312→          new Uint8Array(atob(b64).split("").map((c) => c.charCodeAt(0)));
   313→        unicodeFonts = {
   314→          regular: await pdfDoc.embedFont(decode(NOTO_SANS_REGULAR)),
   315→          bold: await pdfDoc.embedFont(decode(NOTO_SANS_BOLD)),
   316→          italic: await pdfDoc.embedFont(decode(NOTO_SANS_ITALIC)),
   317→          boldItalic: await pdfDoc.embedFont(decode(NOTO_SANS_BOLD_ITALIC)),
   318→        };
   319→        return unicodeFonts;
   320→      };
   321→
   322→      /**
   323→       * Convert a point from view space (top-left origin, y down, points)
   324→       * to PDF space (bottom-left origin, y up) for a page with total
   325→       * rotation T. `dims` = unrotated mediabox {width W, height H},
   326→       * `Wv`/`Hv` = rotated view dimensions.
   327→       */
   328→      const viewPointToPdf = (
   329→        vx: number,
   330→        vy: number,
   331→        T: number,
   332→        H: number,
   333→        Wv: number,
   334→        Hv: number
   335→      ): { x: number; y: number } => {
   336→        let px: number;
   337→        let py: number;
   338→        switch (T) {
   339→          case 90:
   340→            px = vy;
   341→            py = Wv - vx;
   342→            break;
   343→          case 180:
   344→            px = Wv - vx;
   345→            py = Hv - vy;
   346→            break;
   347→          case 270:
   348→            px = Hv - vy;
   349→            py = vx;
   350→            break;
   351→          default:
   352→            px = vx;
   353→            py = vy;
   354→        }
   355→        return { x: px, y: H - py };
   356→      };
   357→
   358→      // Stamps — skip if page not in keepSet
   359→      for (const stamp of exportedStamps) {
   360→        try {
   361→          const page = pdfDoc.getPage(stamp.page - 1);
   362→          const { width: pageWidth, height: pageHeight } = page.getSize();
   363→
   364→          const userRotation = pageRotations[stamp.page] || 0;
   365→          const intrinsic = page.getRotation().angle || 0;
   366→          const T = (((intrinsic + userRotation) % 360) + 360) % 360;
   367→          const { width: Wv, height: Hv } = getViewDims(T, {
   368→            width: pageWidth,
   369→            height: pageHeight,
   370→          });
   371→
   372→          const cw = stamp.canvasWidth || Wv;
   373→          const ch = stamp.canvasHeight || Hv;
   374→
   375→          // Item size in view points
   376→          const pw = (stamp.width / cw) * Wv;
   377→          const ph = (stamp.height / ch) * Hv;
   378→
   379→          // Item center in view space → PDF space
   380→          const cvx = ((stamp.x + stamp.width / 2) / cw) * Wv;
   381→          const cvy = ((stamp.y + stamp.height / 2) / ch) * Hv;
   382→          const center = viewPointToPdf(cvx, cvy, T, pageHeight, Wv, Hv);
   383→
   384→          // pdf-lib rotates CCW; canvas rotation is CW → invert,
   385→          // and account for the page's own /Rotate T.
   386→          const phiDeg = T - stamp.rotation;
   387→          const rad = (phiDeg * Math.PI) / 180;
   388→
   389→          // Anchor so the rotated image stays centered on `center`
   390→          const adjX =
   391→            center.x - (pw / 2) * Math.cos(rad) + (ph / 2) * Math.sin(rad);
   392→          const adjY =
   393→            center.y - (pw / 2) * Math.sin(rad) - (ph / 2) * Math.cos(rad);
   394→
   395→          const response = await fetch(stamp.src);
   396→          const imageArrayBuffer = await response.arrayBuffer();
   397→          const imageBytes = new Uint8Array(imageArrayBuffer);
   398→
   399→          let image;
   400→          try {
   401→            image = await pdfDoc.embedPng(imageBytes);
   402→          } catch {
   403→            try {
   404→              image = await pdfDoc.embedJpg(imageBytes);
   405→            } catch {
   406→              console.error("Could not embed stamp image, skipping");
   407→              continue;
   408→            }
   409→          }
   410→
   411→          page.drawImage(image, {
   412→            x: adjX,
   413→            y: adjY,
   414→            width: pw,
   415→            height: ph,
   416→            rotate: degrees(phiDeg),
   417→            opacity: stamp.opacity,
   418→          });
   419→        } catch (err) {
   420→          console.error("Error embedding stamp:", err);
   421→        }
   422→      }
   423→
   424→      // Texts — skip if page not in keepSet
   425→      for (const textItem of exportedTexts) {
   426→        try {
   427→          const page = pdfDoc.getPage(textItem.page - 1);
   428→          const { width: pageWidth, height: pageHeight } = page.getSize();
   429→
   430→          const userRotation = pageRotations[textItem.page] || 0;
   431→          const intrinsic = page.getRotation().angle || 0;
   432→          const T = (((intrinsic + userRotation) % 360) + 360) % 360;
   433→          const { width: Wv, height: Hv } = getViewDims(T, {
   434→            width: pageWidth,
   435→            height: pageHeight,
   436→          });
   437→
   438→          const cw = textItem.canvasWidth || Wv;
   439→          const ch = textItem.canvasHeight || Hv;
   440→
   441→          const scaledFontSize = (textItem.fontSize / ch) * Hv;
   442→          const scaledLetterSpacing = (textItem.letterSpacing / cw) * Wv;
   443→
   444→          let font: import("pdf-lib").PDFFont;
   445→          const uFonts = await getUnicodeFonts();
   446→          if (textItem.bold && textItem.italic)
   447→            font = uFonts.boldItalic as import("pdf-lib").PDFFont;
   448→          else if (textItem.bold)
   449→            font = uFonts.bold as import("pdf-lib").PDFFont;
   450→          else if (textItem.italic)
   451→            font = uFonts.italic as import("pdf-lib").PDFFont;
   452→          else font = uFonts.regular as import("pdf-lib").PDFFont;
   453→
   454→          const color = hexToRgb(textItem.color);
   455→          const pdfColor = color
   456→            ? rgb(color.r, color.g, color.b)
   457→            : rgb(0, 0, 0);
   458→
   459→          // Multiline support
   460→          const lines = textItem.text.split("\n");
   461→
   462→          // Font ascent — distance from baseline to top of glyph.
   463→          // CSS positions text by top of line-box; glyph is centered within line-height.
   464→          // PDF positions text by baseline. So: baseline = top + leading/2 + ascent.
   465→          const fontAscent = font.heightAtSize(scaledFontSize, {
   466→            descender: false,
   467→          });
   468→          // Line height = 1.2 * fontSize (matches canvas CSS lineHeight: 1.2)
   469→          const scaledLineHeight = scaledFontSize * 1.2;
   470→          const leadingHalf = (scaledLineHeight - scaledFontSize) / 2;
   471→
   472→          // pdf-lib CCW angle that matches the canvas CW angle after page rotation
   473→          const phiDeg = T - textItem.rotation;
   474→          const phiRad = (phiDeg * Math.PI) / 180;
   475→
   476→          for (let li = 0; li < lines.length; li++) {
   477→            const line = lines[li];
   478→            if (!line) {
   479→              continue;
   480→            }
   481→
   482→            // Canvas: top of line-box = textItem.y + li * (fontSize * 1.2)
   483→            const lineCanvasTop = textItem.y + li * textItem.fontSize * 1.2;
   484→            const baselineViewY =
   485→              (lineCanvasTop / ch) * Hv + leadingHalf + fontAscent;
   486→
   487→            // pdf-lib has no charSpacing option — emulate letter spacing by
   488→            // drawing glyph runs per character when it is non-zero.
   489→            const hasSpacing = Math.abs(scaledLetterSpacing) > 0.001;
   490→            const chars = hasSpacing ? Array.from(line) : null;
   491→            const charWidths = chars
   492→              ? chars.map((c) => font.widthOfTextAtSize(c, scaledFontSize))
   493→              : null;
   494→            const totalSpacing = charWidths
   495→              ? scaledLetterSpacing * Math.max(0, chars!.length - 1)
   496→              : 0;
   497→            const naturalWidth = font.widthOfTextAtSize(line, scaledFontSize);
   498→            const lineWidth = charWidths
   499→              ? charWidths.reduce((a, b) => a + b, 0) + totalSpacing
   500→              : naturalWidth;
   501→
   502→            // Compute baseline X in view space based on alignment
   503→            const baseViewX = (textItem.x / cw) * Wv;
   504→            const baselineViewX =
   505→              textItem.align === "center"
   506→                ? baseViewX - lineWidth / 2
   507→                : textItem.align === "right"
   508→                ? baseViewX - lineWidth
   509→                : baseViewX;
   510→
   511→            // Baseline start point → PDF space
   512→            const baselinePdf = viewPointToPdf(
   513→              baselineViewX,
   514→              baselineViewY,
   515→              T,
   516→              pageHeight,
   517→              Wv,
   518→              Hv
   519→            );
   520→
   521→            // Treat the line box as (lineWidth × fontSize) centered on
   522→            // baseline + fontSize/2; anchor so rotation keeps center fixed.
   523→            const lineCenterX = baselinePdf.x + lineWidth / 2;
   524→            const lineCenterY = baselinePdf.y + scaledFontSize / 2;
   525→            const textAdjX =
   526→              lineCenterX -
   527→              (lineWidth / 2) * Math.cos(phiRad) +
   528→              (scaledFontSize / 2) * Math.sin(phiRad);
   529→            const textAdjY =
   530→              lineCenterY -
   531→              (lineWidth / 2) * Math.sin(phiRad) -
   532→              (scaledFontSize / 2) * Math.cos(phiRad);
   533→
   534→            const drawOpts = {
   535→              x: textAdjX,
   536→              y: textAdjY,
   537→              size: scaledFontSize,
   538→              font,
   539→              color: pdfColor,
   540→              rotate: degrees(phiDeg),
   541→            };
   542→
   543→            if (charWidths && chars) {
   544→              // Letter-spaced: place each character along the rotated baseline
   545→              let cursorX = textAdjX;
   546→              let cursorY = textAdjY;
   547→              for (let ci = 0; ci < chars.length; ci++) {
   548→                if (chars[ci] !== " ") {
   549→                  page.drawText(chars[ci], { ...drawOpts, x: cursorX, y: cursorY });
   550→                }
   551→                const adv = charWidths[ci] + scaledLetterSpacing;
   552→                cursorX += adv * Math.cos(phiRad);
   553→                cursorY += adv * Math.sin(phiRad);
   554→              }
   555→            } else {
   556→              page.drawText(line, drawOpts);
   557→            }
   558→
   559→            // Underline: thin rectangle below the baseline, offset
   560→            // perpendicular to the text direction so it survives rotation.
   561→            if (textItem.underline) {
   562→              const underlineThickness = Math.max(0.5, scaledFontSize / 18);
   563→              const off = scaledFontSize * 0.12;
   564→              page.drawRectangle({
   565→                x: textAdjX + off * Math.sin(phiRad),
   566→                y: textAdjY - off * Math.cos(phiRad),
   567→                width: lineWidth,
   568→                height: underlineThickness,
   569→                color: pdfColor,
   570→                rotate: degrees(phiDeg),
   571→              });
   572→            }
   573→          }
   574→        } catch (err) {
   575→          console.error("Error drawing text:", err);
   576→        }
   577→      }
   578→
   579→      // Erasers — skip if page not in keepSet
   580→      for (const eraserItem of exportedErasers) {
   581→        try {
   582→          const page = pdfDoc.getPage(eraserItem.page - 1);
   583→          const { width: pageWidth, height: pageHeight } = page.getSize();
   584→
   585→          const userRotation = pageRotations[eraserItem.page] || 0;
   586→          const intrinsic = page.getRotation().angle || 0;
   587→          const T = (((intrinsic + userRotation) % 360) + 360) % 360;
   588→          const { width: Wv, height: Hv } = getViewDims(T, {
   589→            width: pageWidth,
   590→            height: pageHeight,
   591→          });
   592→
   593→          const cw = eraserItem.canvasWidth || Wv;
   594→          const ch = eraserItem.canvasHeight || Hv;
   595→
   596→          const eraserColor = hexToRgb(eraserItem.color);
   597→          const pdfColor = eraserColor
   598→            ? rgb(eraserColor.r, eraserColor.g, eraserColor.b)
   599→            : rgb(1, 1, 1);
   600→
   601→          const pdfStrokeWidth = (eraserItem.strokeWidth / ch) * Hv;
   602→
   603→          const pdfPoints = eraserItem.points.map((p) => {
   604→            const vx = (p.x / cw) * Wv;
   605→            const vy = (p.y / ch) * Hv;
   606→            return viewPointToPdf(vx, vy, T, pageHeight, Wv, Hv);
   607→          });
   608→
   609→          for (let i = 0; i < pdfPoints.length; i++) {
   610→            const p = pdfPoints[i];
   611→
   612→            if (i === 0 && pdfPoints.length === 1) {
   613→              const halfSize = pdfStrokeWidth / 2;
   614→              page.drawRectangle({
   615→                x: p.x - halfSize,
   616→                y: p.y - halfSize,
   617→                width: pdfStrokeWidth,
   618→                height: pdfStrokeWidth,
   619→                color: pdfColor,
   620→              });
   621→            } else if (i > 0) {
   622→              const prevP = pdfPoints[i - 1];
   623→
   624→              page.drawLine({
   625→                start: { x: prevP.x, y: prevP.y },
   626→                end: { x: p.x, y: p.y },
   627→                thickness: pdfStrokeWidth,
   628→                color: pdfColor,
   629→              });
   630→            }
   631→          }
   632→        } catch (err) {
   633→          console.error("Error drawing eraser:", err);
   634→        }
   635→      }
   636→
   637→      // Remove pages that are not in keepSet (deleted or not selected for export)
   638→      // Iterate from the highest index to the lowest to avoid shifting issues.
   639→      const indicesToRemove: number[] = [];
   640→      for (let i = 0; i < pageCount; i++) {
   641→        if (!keepSet.has(i + 1)) indicesToRemove.push(i);
   642→      }
   643→      // Sort descending so removal doesn't shift subsequent indices
   644→      indicesToRemove.sort((a, b) => b - a);
   645→      for (const idx of indicesToRemove) {
   646→        try {
   647→          pdfDoc.removePage(idx);
   648→        } catch (err) {
   649→          console.error(`Error removing page ${idx + 1}:`, err);
   650→        }
   651→      }
   652→
   653→      const modifiedPdfBytes = await pdfDoc.save();
   654→      const blob = new Blob([new Uint8Array(modifiedPdfBytes)], {
   655→        type: "application/pdf",
   656→      });
   657→      const url = URL.createObjectURL(blob);
   658→
   659→      const link = document.createElement("a");
   660→      link.href = url;
   661→      link.download = pdfFileName
   662→        ? `modified_${pdfFileName}`
   663→        : "modified_document.pdf";
   664→      document.body.appendChild(link);
   665→      link.click();
   666→      document.body.removeChild(link);
   667→
   668→      setTimeout(() => URL.revokeObjectURL(url), 5000);
   669→
   670→      toast.dismiss(loadingToast);
   671→      toast.success("PDF сохранён!", {
   672→        description: `${pagesToKeep.length} стр. · ${exportedStamps.length} печатей · ${exportedTexts.length} текстов · ${exportedErasers.length} мазков`,
   673→      });
   674→    } catch (error) {
   675→      toast.dismiss(loadingToast);
   676→      console.error("Error downloading PDF:", error);
   677→      toast.error("Ошибка при сохранении PDF", {
   678→        description: String(error),
   679→      });
   680→    } finally {
   681→      setIsDownloading(false);
   682→    }
   683→  }, [
   684→    pdfArrayBuffer,
   685→    stamps,
   686→    texts,
   687→    erasers,
   688→    pdfFileName,
   689→    isDownloading,
   690→  ]);
   691→
   692→  const toolbarContent = (
   693→    <Toolbar
   694→      onUploadClick={() => {
   695→        handleUploadClick();
   696→        setSidebarOpen(false);
   697→      }}
   698→      onDownloadClick={() => {
   699→        setExportDialogOpen(true);
   700→        setSidebarOpen(false);
   701→      }}
   702→      isDownloading={isDownloading}
   703→    />
   704→  );
   705→
   706→  if (!authenticated) {
   707→    return <LoginScreen onLogin={handleLogin} />;
   708→  }
   709→
   710→  return (
   711→    <div className="min-h-screen flex flex-col bg-background">
   712→      {/* Header */}
   713→      <header className="border-b border-border/70 glass-strong px-4 py-2.5 flex items-center justify-between shrink-0 z-30">
   714→        <div className="flex items-center gap-3">
   715→          {pdfFile && (
   716→            <Sheet open={sidebarOpen} onOpenChange={setSidebarOpen}>
   717→              <SheetTrigger asChild>
   718→                <Button
   719→                  variant="ghost"
   720→                  size="icon"
   721→                  className="md:hidden h-9 w-9 rounded-xl"
   722→                >
   723→                  <Menu className="h-5 w-5" />
   724→                </Button>
   725→              </SheetTrigger>
   726→              <SheetContent side="left" className="p-0 w-72 overflow-y-auto">
   727→                <SheetTitle className="sr-only">Панель инструментов</SheetTitle>
   728→                {toolbarContent}
   729→                <div className="h-px bg-border/60 mx-4" />
   730→                <LayersPanel currentPage={currentPage} totalPages={totalPages} />
   731→              </SheetContent>
   732→            </Sheet>
   733→          )}
   734→
   735→          {/* Logo */}
   736→          <div className="relative h-9 w-9 rounded-xl bg-ink flex items-center justify-center shadow-soft">
   737→            <FileText
   738→              className="h-5 w-5 text-white"
   739→              strokeWidth={2}
   740→              fill="rgba(255,255,255,0.15)"
   741→            />
   742→          </div>
   743→          <div>
   744→            <h1 className="text-base font-semibold leading-tight tracking-tight">
   745→              PDF{" "}
   746→              <span className="text-primary">Редактор</span>
   747→            </h1>
   748→            <p className="text-[11px] text-muted-foreground hidden sm:block">
   749→              Печати и текст на документах
   750→            </p>
   751→          </div>
   752→          {pdfFile && (
   753→            <div className="hidden lg:flex items-center gap-1.5 ml-6">
   754→              {[
   755→                { kbd: "Ctrl+Колесо", label: "масштаб" },
   756→                { kbd: "← →", label: "страницы" },
   757→                { kbd: "Del", label: "удалить" },
   758→                { kbd: "2× клик", label: "текст" },
   759→              ].map((h) => (
   760→                <span
   761→                  key={h.kbd}
   762→                  className="inline-flex items-center gap-1.5 px-2 py-1 rounded-lg bg-secondary/60 border border-border/50 hover:border-primary/30 transition-colors"
   763→                >
   764→                  <kbd className="px-1.5 py-0.5 bg-card rounded text-[10px] font-mono border border-border/70 shadow-soft">
   765→                    {h.kbd}
   766→                  </kbd>
   767→                  <span className="text-[11px] text-muted-foreground">{h.label}</span>
   768→                </span>
   769→              ))}
   770→            </div>
   771→          )}
   772→        </div>
   773→        <div className="flex items-center gap-2">
   774→          {pdfFile && (
   775→            <div className="hidden sm:flex items-center gap-2 px-3 py-1.5 rounded-xl bg-secondary/70 border border-border/50 max-w-[220px]">
   776→              <div className="h-6 w-6 rounded-md bg-primary/15 flex items-center justify-center shrink-0">
   777→                <FileText className="h-3.5 w-3.5 text-primary" />
   778→              </div>
   779→              <span className="text-xs text-muted-foreground truncate">
   780→                {pdfFileName}
   781→              </span>
   782→            </div>
   783→          )}
   784→          <Button
   785→            variant="ghost"
   786→            size="sm"
   787→            className="gap-1.5 text-muted-foreground hover:text-foreground h-9 rounded-xl font-medium"
   788→            onClick={handleLogout}
   789→          >
   790→            <LogOut className="h-4 w-4" />
   791→            <span className="hidden sm:inline">Выход</span>
   792→          </Button>
   793→        </div>
   794→      </header>
   795→
   796→      {/* Main content */}
   797→      <div className="flex-1 flex overflow-hidden">
   798→        {pdfFile && (
   799→          <aside className="hidden md:block w-72 border-r border-border/70 bg-card/30 overflow-y-auto shrink-0">
   800→            {toolbarContent}
   801→            <div className="h-px bg-border/60 mx-4" />
   802→            <LayersPanel currentPage={currentPage} totalPages={totalPages} />
   803→          </aside>
   804→        )}
   805→
   806→        {pdfFile ? <PdfCanvas /> : <UploadZone />}
   807→      </div>
   808→
   809→      {/* Footer */}
   810→      <footer className="border-t border-border/70 glass-strong px-4 py-2.5 text-center text-xs text-muted-foreground shrink-0 mt-auto">
   811→        <div className="flex items-center justify-center gap-2">
   812→          <ShieldCheck className="h-3 w-3 text-primary" />
   813→          <span className="font-medium">
   814→            PDF Редактор · документы не покидают ваш браузер
   815→          </span>
   816→        </div>
   817→      </footer>
   818→
   819→      <input
   820→        ref={fileInputRef}
   821→        type="file"
   822→        accept=".pdf,application/pdf"
   823→        onChange={handleFileChange}
   824→        className="hidden"
   825→      />
   826→
   827→      {/* Export dialog */}
   828→      <ExportDialog
   829→        open={exportDialogOpen}
   830→        onOpenChange={setExportDialogOpen}
   831→        onExport={handleDownload}
   832→      />
   833→    </div>
   834→  );
   835→}

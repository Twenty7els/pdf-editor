     1→"use client";
     2→
     3→/**
     4→ * Экран «Автозаполнение документов»:
     5→ *  1. Загрузка заполненной анкеты (.xlsx «Заявка на подключение»)
     6→ *  2. Редактирование извлечённого профиля мерчанта
     7→ *  3. Выбор целевого документа → генерация и скачивание
     8→ */
     9→import React, { useCallback, useEffect, useMemo, useRef, useState } from "react";
    10→import { toast } from "sonner";
    11→import {
    12→  AlertCircle,
    13→  Check,
    14→  CircleDashed,
    15→  Download,
    16→  FileSpreadsheet,
    17→  FileText,
    18→  ListChecks,
    19→  Loader2,
    20→  RefreshCw,
    21→  Sparkles,
    22→  Upload,
    23→} from "lucide-react";
    24→import { Button } from "@/components/ui/button";
    25→import {
    26→  PROFILE_GROUPS,
    27→  type MerchantProfile,
    28→} from "@/lib/doc-autofill/profile";
    29→import {
    30→  TARGET_REQUIREMENTS,
    31→  isRequirementSatisfied,
    32→  missingForTarget,
    33→  type TemplateRequirement,
    34→} from "@/lib/doc-autofill/requirements";
    35→
    36→interface TargetInfo {
    37→  id: string;
    38→  title: string;
    39→  description: string;
    40→  ext: string;
    41→  optional: boolean;
    42→  available: boolean;
    43→}
    44→
    45→const LIST_KEYS = new Set(["serials", "pointAddresses", "pointComments"]);
    46→
    47→export default function DocAutofillApp() {
    48→  const [profile, setProfile] = useState<MerchantProfile | null>(null);
    49→  const [sourceName, setSourceName] = useState("");
    50→  const [warnings, setWarnings] = useState<string[]>([]);
    51→  const [parsing, setParsing] = useState(false);
    52→  const [targets, setTargets] = useState<TargetInfo[]>([]);
    53→  const [selected, setSelected] = useState<string>("");
    54→  const [generating, setGenerating] = useState(false);
    55→  const [uploadingTemplate, setUploadingTemplate] = useState<string | null>(
    56→    null
    57→  );
    58→  const anketaInputRef = useRef<HTMLInputElement>(null);
    59→  const templateInputRef = useRef<HTMLInputElement>(null);
    60→  const templateTargetRef = useRef<string>("");
    61→  const checklistRef = useRef<HTMLDivElement>(null);
    62→
    63→  /** Поля, которые пользователь сейчас/только что редактировал в чек-листе:
    64→   *  пока поле закреплено, инпут не исчезает при вводе и зелёный чип не появляется. */
    65→  const [pinned, setPinned] = useState<Set<string>>(new Set());
    66→  const pin = useCallback((k: string) => {
    67→    setPinned((p) => (p.has(k) ? p : new Set(p).add(k)));
    68→  }, []);
    69→  const unpin = useCallback((k: string) => {
    70→    setPinned((p) => {
    71→      if (!p.has(k)) return p;
    72→      const n = new Set(p);
    73→      n.delete(k);
    74→      return n;
    75→    });
    76→  }, []);
    77→
    78→  const selectedTarget = targets.find((t) => t.id === selected) ?? null;
    79→  const requirements: TemplateRequirement[] = useMemo(
    80→    () => (selected && profile ? TARGET_REQUIREMENTS[selected] ?? [] : []),
    81→    [selected, profile]
    82→  );
    83→  const missing = useMemo(
    84→    () => (profile && selected ? missingForTarget(selected, profile) : []),
    85→    [profile, selected]
    86→  );
    87→  const present = useMemo(
    88→    () =>
    89→      requirements.filter(
    90→        (r) =>
    91→          profile &&
    92→          !pinned.has(String(r.key)) &&
    93→          isRequirementSatisfied(r, profile)
    94→      ),
    95→    [requirements, profile, pinned]
    96→  );
    97→  /** Инпуты чек-листа: незаполненные + закреплённые за курсором (чтобы не исчезали при вводе). */
    98→  const gridReqs = useMemo(() => {
    99→    const keys = new Set(missing.map((r) => String(r.key)));
   100→    pinned.forEach((k) => keys.add(k));
   101→    return requirements.filter((r) => r.key && keys.has(String(r.key)));
   102→  }, [requirements, missing, pinned]);
   103→  /** Автополя, которые пока не собраны — показываем как подсказку, без инпута. */
   104→  const autoPending = useMemo(
   105→    () =>
   106→      requirements.filter(
   107→        (r) => r.autoNote && profile && !isRequirementSatisfied(r, profile)
   108→      ),
   109→    [requirements, profile]
   110→  );
   111→
   112→  const loadTargets = useCallback(async () => {
   113→    try {
   114→      const res = await fetch("/api/doc-autofill/templates");
   115→      const data = await res.json();
   116→      setTargets(data.targets ?? []);
   117→      setSelected((prev) => {
   118→        if (prev) return prev;
   119→        const first = (data.targets ?? []).find(
   120→          (t: TargetInfo) => t.available
   121→        );
   122→        return first ? first.id : "";
   123→      });
   124→    } catch {
   125→      toast.error("Не удалось загрузить список шаблонов");
   126→    }
   127→  }, []);
   128→
   129→  useEffect(() => {
   130→    void loadTargets();
   131→  }, [loadTargets]);
   132→
   133→  const handleAnketaFile = useCallback(async (file: File) => {
   134→    if (!file.name.toLowerCase().endsWith(".xlsx")) {
   135→      toast.error("Анкета должна быть в формате .xlsx");
   136→      return;
   137→    }
   138→    setParsing(true);
   139→    try {
   140→      const fd = new FormData();
   141→      fd.append("file", file);
   142→      const res = await fetch("/api/doc-autofill/parse", {
   143→        method: "POST",
   144→        body: fd,
   145→      });
   146→      const data = await res.json();
   147→      if (!res.ok) {
   148→        toast.error(data.error ?? "Ошибка разбора анкеты");
   149→        return;
   150→      }
   151→      setProfile(data.profile);
   152→      setSourceName(data.sourceName ?? file.name);
   153→      setWarnings(data.warnings ?? []);
   154→      toast.success("Анкета распознана", {
   155→        description: file.name,
   156→      });
   157→    } catch (err) {
   158→      console.error(err);
   159→      toast.error("Ошибка загрузки файла");
   160→    } finally {
   161→      setParsing(false);
   162→    }
   163→  }, []);
   164→
   165→  const handleGenerate = useCallback(async () => {
   166→    if (!profile || !selected || generating) return;
   167→    const target = targets.find((t) => t.id === selected);
   168→    if (!target?.available) {
   169→      toast.error("Шаблон не загружен");
   170→      return;
   171→    }
   172→    setGenerating(true);
   173→    try {
   174→      const res = await fetch("/api/doc-autofill/generate", {
   175→        method: "POST",
   176→        headers: { "Content-Type": "application/json" },
   177→        body: JSON.stringify({ targetId: selected, profile }),
   178→      });
   179→      if (!res.ok) {
   180→        const data = await res.json().catch(() => ({}));
   181→        toast.error(data.error ?? "Ошибка генерации документа");
   182→        return;
   183→      }
   184→      const blob = await res.blob();
   185→      const url = URL.createObjectURL(blob);
   186→      const link = document.createElement("a");
   187→      link.href = url;
   188→      const cd = res.headers.get("Content-Disposition") ?? "";
   189→      const m = cd.match(/filename="([^"]+)"/);
   190→      link.download = m?.[1] ?? `document.${target.ext}`;
   191→      document.body.appendChild(link);
   192→      link.click();
   193→      document.body.removeChild(link);
   194→      setTimeout(() => URL.revokeObjectURL(url), 5000);
   195→      toast.success("Документ сформирован", {
   196→        description: target.title,
   197→      });
   198→    } catch (err) {
   199→      console.error(err);
   200→      toast.error("Ошибка генерации документа");
   201→    } finally {
   202→      setGenerating(false);
   203→    }
   204→  }, [profile, selected, generating, targets]);
   205→
   206→  const handleTemplateUpload = useCallback(
   207→    async (file: File) => {
   208→      const targetId = templateTargetRef.current;
   209→      if (!targetId) return;
   210→      setUploadingTemplate(targetId);
   211→      try {
   212→        const fd = new FormData();
   213→        fd.append("file", file);
   214→        fd.append("targetId", targetId);
   215→        const res = await fetch("/api/doc-autofill/templates", {
   216→          method: "POST",
   217→          body: fd,
   218→        });
   219→        const data = await res.json();
   220→        if (!res.ok) {
   221→          toast.error(data.error ?? "Не удалось загрузить шаблон");
   222→          return;
   223→        }
   224→        toast.success("Шаблон загружен");
   225→        await loadTargets();
   226→      } catch {
   227→        toast.error("Не удалось загрузить шаблон");
   228→      } finally {
   229→        setUploadingTemplate(null);
   230→      }
   231→    },
   232→    [loadTargets]
   233→  );
   234→
   235→  const setField = (key: keyof MerchantProfile, value: unknown) => {
   236→    setProfile((p) => (p ? { ...p, [key]: value } : p));
   237→  };
   238→
   239→  // при смене целевого документа сбрасываем закрепления инпутов
   240→  useEffect(() => {
   241→    setPinned(new Set());
   242→  }, [selected]);
   243→
   244→  const filledCount = profile
   245→    ? Object.entries(profile).filter(([k, v]) =>
   246→        LIST_KEYS.has(k)
   247→          ? (v as string[]).length > 0
   248→          : Boolean(String(v ?? "").trim())
   249→      ).length
   250→    : 0;
   251→
   252→  return (
   253→    <div className="flex-1 overflow-y-auto stage-bg">
   254→      <div className="max-w-5xl mx-auto px-4 py-8 md:py-10 w-full">
   255→        {/* Hero */}
   256→        <div className="text-center mb-8 stagger-item">
   257→          <div className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full bg-card/80 border border-border/60 text-xs text-muted-foreground mb-4">
   258→            <Sparkles className="h-3.5 w-3.5 text-terracotta" />
   259→            Данные подставляются по смыслу
   260→          </div>
   261→          <h2 className="display-title text-3xl md:text-4xl">
   262→            Автозаполнение{" "}
   263→            <span className="text-terracotta-dark">документов</span>
   264→          </h2>
   265→          <p className="text-sm text-muted-foreground mt-2 max-w-xl mx-auto">
   266→            Загрузите заполненную анкету мерчанта — данные автоматически
   267→            встанут в заявку на регистрацию СБП или анкету-заявление ИП/ЮЛ.
   268→          </p>
   269→        </div>
   270→
   271→        <div className="space-y-5">
   272→          {/* Шаг 1 — анкета */}
   273→          <section
   274→            className="bg-card border border-border/70 rounded-2xl shadow-paper p-5 md:p-6 stagger-item"
   275→            style={{ ["--stagger-delay" as string]: "60ms" }}
   276→          >
   277→            <div className="flex items-start justify-between gap-3 mb-4">
   278→              <div>
   279→                <div className="text-[11px] uppercase tracking-widest text-muted-foreground font-medium">
   280→                  Шаг 1
   281→                </div>
   282→                <h3 className="display-title text-xl mt-0.5">
   283→                  Заполненная анкета
   284→                </h3>
   285→              </div>
   286→              {profile && (
   287→                <Button
   288→                  variant="outline"
   289→                  size="sm"
   290→                  className="gap-1.5 rounded-xl"
   291→                  onClick={() => anketaInputRef.current?.click()}
   292→                >
   293→                  <RefreshCw className="h-3.5 w-3.5" />
   294→                  Другая
   295→                </Button>
   296→              )}
   297→            </div>
   298→
   299→            {!profile ? (
   300→              <button
   301→                type="button"
   302→                onClick={() => anketaInputRef.current?.click()}
   303→                disabled={parsing}
   304→                className="w-full rounded-xl border-2 border-dashed border-border bg-secondary/30 hover:bg-secondary/50 transition-colors p-8 flex flex-col items-center gap-3 text-center disabled:opacity-60"
   305→              >
   306→                {parsing ? (
   307→                  <Loader2 className="h-8 w-8 text-terracotta animate-spin" />
   308→                ) : (
   309→                  <FileSpreadsheet className="h-8 w-8 text-muted-foreground" />
   310→                )}
   311→                <div>
   312→                  <div className="font-medium text-sm">
   313→                    {parsing
   314→                      ? "Разбираем анкету…"
   315→                      : "Нажмите, чтобы выбрать .xlsx анкету"}
   316→                  </div>
   317→                  <div className="text-xs text-muted-foreground mt-1">
   318→                    «Заявка на подключение» (Uniteller) после заполнения
   319→                  </div>
   320→                </div>
   321→              </button>
   322→            ) : (
   323→              <div className="flex items-center gap-3 p-3.5 rounded-xl bg-oat/60 border border-border/50">
   324→                <div className="h-10 w-10 rounded-xl bg-emerald-100 flex items-center justify-center shrink-0">
   325→                  <Check className="h-5 w-5 text-emerald-700" />
   326→                </div>
   327→                <div className="min-w-0 flex-1">
   328→                  <div className="text-sm font-medium truncate">{sourceName}</div>
   329→                  <div className="text-xs text-muted-foreground">
   330→                    Распознано полей: {filledCount}
   331→                  </div>
   332→                </div>
   333→              </div>
   334→            )}
   335→
   336→            {warnings.length > 0 && profile && (
   337→              <div className="mt-3 rounded-xl bg-amber-50 border border-amber-200/70 p-3 space-y-1.5">
   338→                {warnings.map((w, i) => (
   339→                  <div
   340→                    key={i}
   341→                    className="flex items-start gap-2 text-xs text-amber-800"
   342→                  >
   343→                    <AlertCircle className="h-3.5 w-3.5 mt-0.5 shrink-0" />
   344→                    {w}
   345→                  </div>
   346→                ))}
   347→              </div>
   348→            )}
   349→          </section>
   350→
   351→          {/* Шаг 2 — профиль */}
   352→          {profile && (
   353→            <section
   354→              className="bg-card border border-border/70 rounded-2xl shadow-paper p-5 md:p-6 stagger-item"
   355→              style={{ ["--stagger-delay" as string]: "120ms" }}
   356→            >
   357→              <div className="mb-4">
   358→                <div className="text-[11px] uppercase tracking-widest text-muted-foreground font-medium">
   359→                  Шаг 2
   360→                </div>
   361→                <h3 className="display-title text-xl mt-0.5">
   362→                  Данные мерчанта
   363→                </h3>
   364→                <p className="text-xs text-muted-foreground mt-1">
   365→                  Проверьте и при необходимости поправьте — в документы встанут
   366→                  именно эти значения.
   367→                </p>
   368→              </div>
   369→
   370→              <div className="grid md:grid-cols-2 gap-4">
   371→                {PROFILE_GROUPS.map((group) => (
   372→                  <div
   373→                    key={group.title}
   374→                    className="rounded-xl border border-border/50 bg-secondary/20 p-4"
   375→                  >
   376→                    <div className="text-xs font-semibold text-terracotta-dark uppercase tracking-wide mb-3">
   377→                      {group.title}
   378→                    </div>
   379→                    <div className="space-y-2.5">
   380→                      {group.fields.map((f) => (
   381→                        <div key={String(f.key)}>
   382→                          <label className="text-[11px] text-muted-foreground block mb-1">
   383→                            {f.label}
   384→                          </label>
   385→                          {f.list ? (
   386→                            <textarea
   387→                              value={((profile[f.key] as string[]) ?? []).join(
   388→                                "\n"
   389→                              )}
   390→                              onChange={(e) =>
   391→                                setField(
   392→                                  f.key,
   393→                                  e.target.value
   394→                                    .split("\n")
   395→                                    .map((s) => s.trim())
   396→                                    .filter(Boolean)
   397→                                )
   398→                              }
   399→                              rows={Math.min(
   400→                                4,
   401→                                Math.max(
   402→                                  2,
   403→                                  ((profile[f.key] as string[]) ?? []).length ||
   404→                                    2
   405→                                )
   406→                              )}
   407→                              placeholder="По одному в строке"
   408→                              className="w-full text-sm rounded-lg border border-border/60 bg-card px-3 py-2 focus:outline-none focus:ring-2 focus:ring-terracotta/30 focus:border-terracotta/50 resize-y min-h-[60px] max-h-48"
   409→                            />
   410→                          ) : (
   411→                            <input
   412→                              value={String(profile[f.key] ?? "")}
   413→                              onChange={(e) =>
   414→                                setField(f.key, e.target.value)
   415→                              }
   416→                              className="w-full text-sm rounded-lg border border-border/60 bg-card px-3 py-2 focus:outline-none focus:ring-2 focus:ring-terracotta/30 focus:border-terracotta/50"
   417→                            />
   418→                          )}
   419→                        </div>
   420→                      ))}
   421→                    </div>
   422→                  </div>
   423→                ))}
   424→              </div>
   425→            </section>
   426→          )}
   427→
   428→          {/* Шаг 3 — целевой документ */}
   429→          {profile && (
   430→            <section
   431→              className="bg-card border border-border/70 rounded-2xl shadow-paper p-5 md:p-6 stagger-item"
   432→              style={{ ["--stagger-delay" as string]: "180ms" }}
   433→            >
   434→              <div className="mb-4">
   435→                <div className="text-[11px] uppercase tracking-widest text-muted-foreground font-medium">
   436→                  Шаг 3
   437→                </div>
   438→                <h3 className="display-title text-xl mt-0.5">
   439→                  Куда подставить данные
   440→                </h3>
   441→              </div>
   442→
   443→              <div className="grid sm:grid-cols-3 gap-3">
   444→                {targets.map((t) => {
   445→                  const active = selected === t.id && t.available;
   446→                  return (
   447→                    <div key={t.id} className="relative">
   448→                      <button
   449→                        type="button"
   450→                        disabled={!t.available}
   451→                        onClick={() => {
   452→                          setSelected(t.id);
   453→                          // панель проверки ниже — плавно показать её пользователю
   454→                          setTimeout(
   455→                            () =>
   456→                              checklistRef.current?.scrollIntoView({
   457→                                behavior: "smooth",
   458→                                block: "nearest",
   459→                              }),
   460→                            60
   461→                          );
   462→                        }}
   463→                        className={`w-full text-left rounded-xl border p-4 transition-all ${
   464→                          active
   465→                            ? "border-terracotta bg-terracotta/5 shadow-soft ring-2 ring-terracotta/20"
   466→                            : t.available
   467→                            ? "border-border/60 bg-card hover:border-terracotta/40 hover:shadow-soft"
   468→                            : "border-border/40 bg-secondary/20 opacity-70 cursor-default"
   469→                        }`}
   470→                      >
   471→                        <div className="flex items-start justify-between gap-2">
   472→                          <div
   473→                            className={`h-9 w-9 rounded-lg flex items-center justify-center ${
   474→                              active
   475→                                ? "bg-terracotta text-white"
   476→                                : "bg-secondary/70 text-muted-foreground"
   477→                            }`}
   478→                          >
   479→                            {t.ext === "xlsx" ? (
   480→                              <FileSpreadsheet className="h-4 w-4" />
   481→                            ) : (
   482→                              <FileText className="h-4 w-4" />
   483→                            )}
   484→                          </div>
   485→                          {active && (
   486→                            <div className="h-5 w-5 rounded-full bg-terracotta flex items-center justify-center">
   487→                              <Check className="h-3 w-3 text-white" />
   488→                            </div>
   489→                          )}
   490→                        </div>
   491→                        <div className="font-medium text-sm mt-2.5">
   492→                          {t.title}
   493→                        </div>
   494→                        <div className="text-[11px] text-muted-foreground mt-1 leading-relaxed">
   495→                          {t.description}
   496→                        </div>
   497→                        {!t.available && (
   498→                          <div className="mt-2 text-[11px] font-medium text-amber-700">
   499→                            Шаблон не загружен
   500→                          </div>
   501→                        )}
   502→                      </button>
   503→
   504→                      {!t.available && (
   505→                        <button
   506→                          type="button"
   507→                          onClick={() => {
   508→                            templateTargetRef.current = t.id;
   509→                            templateInputRef.current?.click();
   510→                          }}
   511→                          disabled={uploadingTemplate === t.id}
   512→                          className="absolute inset-x-4 bottom-4 inline-flex items-center justify-center gap-1.5 text-xs font-medium rounded-lg border border-amber-300/70 bg-amber-50 hover:bg-amber-100 text-amber-800 py-1.5 transition-colors disabled:opacity-60"
   513→                        >
   514→                          {uploadingTemplate === t.id ? (
   515→                            <Loader2 className="h-3.5 w-3.5 animate-spin" />
   516→                          ) : (
   517→                            <Upload className="h-3.5 w-3.5" />
   518→                          )}
   519→                          Загрузить шаблон
   520→                        </button>
   521→                      )}
   522→                    </div>
   523→                  );
   524→                })}
   525→              </div>
   526→
   527→              {/* Проверка данных выбранного шаблона */}
   528→              {selectedTarget && selectedTarget.available && requirements.length > 0 && (
   529→                <div
   530→                  ref={checklistRef}
   531→                  className="mt-4 rounded-xl border border-border/60 bg-secondary/25 p-4"
   532→                >
   533→                  <div className="flex items-center justify-between gap-2 flex-wrap mb-3">
   534→                    <div className="flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wide text-terracotta-dark">
   535→                      <ListChecks className="h-3.5 w-3.5" />

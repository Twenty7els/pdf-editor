     9→     9→import React, { useCallback, useEffect, useMemo, useRef, useState } from "react";
    10→    10→import { toast } from "sonner";

    24→    24→import { Button } from "@/components/ui/button";

    46→    46→
    47→    47→export default function DocAutofillApp() {
    48→    48→  const [profile, setProfile] = useState<MerchantProfile | null>(null);
    49→    49→  const [sourceName, setSourceName] = useState("");
    50→    50→  const [warnings, setWarnings] = useState<string[]>([]);
    51→    51→  const [parsing, setParsing] = useState(false);
    52→    52→  const [targets, setTargets] = useState<TargetInfo[]>([]);
    53→    53→  const [selected, setSelected] = useState<string>("");
    54→    54→  const [generating, setGenerating] = useState(false);
    55→    55→  const [uploadingTemplate, setUploadingTemplate] = useState<string | null>(
    56→    56→    null
    57→    57→  );
    58→    58→  const anketaInputRef = useRef<HTMLInputElement>(null);
    59→    59→  const templateInputRef = useRef<HTMLInputElement>(null);
    60→    60→  const templateTargetRef = useRef<string>("");
    61→    61→  const checklistRef = useRef<HTMLDivElement>(null);
    62→    62→
    63→    63→  /** Поля, которые пользователь сейчас/только что редактировал в чек-листе:
    64→    64→   *  пока поле закреплено, инпут не исчезает при вводе и зелёный чип не появляется. */
    65→    65→  const [pinned, setPinned] = useState<Set<string>>(new Set());
    66→    66→  const pin = useCallback((k: string) => {

    75→    75→    });
    76→    76→  }, []);
    77→    77→
    78→    78→  const selectedTarget = targets.find((t) => t.id === selected) ?? null;
    79→    79→  const requirements: TemplateRequirement[] = useMemo(
    80→    80→    () => (selected && profile ? TARGET_REQUIREMENTS[selected] ?? [] : []),
    81→    81→    [selected, profile]
    82→    82→  );
    83→    83→  const missing = useMemo(
    84→    84→    () => (profile && selected ? missingForTarget(selected, profile) : []),
    85→    85→    [profile, selected]
    86→    86→  );
    87→    87→  const present = useMemo(
    88→    88→    () =>
    89→    89→      requirements.filter(
    90→    90→        (r) =>
    91→    91→          profile &&
    92→    92→          !pinned.has(String(r.key)) &&
    93→    93→          isRequirementSatisfied(r, profile)
    94→    94→      ),
    95→    95→    [requirements, profile, pinned]
    96→    96→  );

    99→    99→    const keys = new Set(missing.map((r) => String(r.key)));
   100→   100→    pinned.forEach((k) => keys.add(k));
   101→   101→    return requirements.filter((r) => r.key && keys.has(String(r.key)));
   102→   102→  }, [requirements, missing, pinned]);

   105→   105→    () =>
   106→   106→      requirements.filter(
   107→   107→        (r) => r.autoNote && profile && !isRequirementSatisfied(r, profile)

   117→   117→      setSelected((prev) => {

   119→   119→        const first = (data.targets ?? []).find(
   120→   120→          (t: TargetInfo) => t.available
   121→   121→        );
   122→   122→        return first ? first.id : "";
   123→   123→      });

   125→   125→      toast.error("Не удалось загрузить список шаблонов");
   126→   126→    }
   127→   127→  }, []);
   128→   128→
   129→   129→  useEffect(() => {
   130→   130→    void loadTargets();
   131→   131→  }, [loadTargets]);

   135→   135→      toast.error("Анкета должна быть в формате .xlsx");
   136→   136→      return;
   137→   137→    }

   139→   139→    try {
   140→   140→      const fd = new FormData();
   141→   141→      fd.append("file", file);
   142→   142→      const res = await fetch("/api/doc-autofill/parse", {
   143→   143→        method: "POST",
   144→   144→        body: fd,
   145→   145→      });
   146→   146→      const data = await res.json();
   147→   147→      if (!res.ok) {
   148→   148→        toast.error(data.error ?? "Ошибка разбора анкеты");
   149→   149→        return;
   150→   150→      }
   151→   151→      setProfile(data.profile);
   152→   152→      setSourceName(data.sourceName ?? file.name);
   153→   153→      setWarnings(data.warnings ?? []);
   154→   154→      toast.success("Анкета распознана", {
   155→   155→        description: file.name,
   156→   156→      });
   157→   157→    } catch (err) {
   158→   158→      console.error(err);
   159→   159→      toast.error("Ошибка загрузки файла");
   160→   160→    } finally {
   161→   161→      setParsing(false);
   162→   162→    }
   163→   163→  }, []);
   164→   164→
   165→   165→  const handleGenerate = useCallback(async () => {
   166→   166→    if (!profile || !selected || generating) return;
   167→   167→    const target = targets.find((t) => t.id === selected);
   168→   168→    if (!target?.available) {
   169→   169→      toast.error("Шаблон не загружен");
   170→   170→      return;
   171→   171→    }
   172→   172→    setGenerating(true);
   173→   173→    try {
   174→   174→      const res = await fetch("/api/doc-autofill/generate", {
   175→   175→        method: "POST",
   176→   176→        headers: { "Content-Type": "application/json" },
   177→   177→        body: JSON.stringify({ targetId: selected, profile }),
   178→   178→      });
   179→   179→      if (!res.ok) {
   180→   180→        const data = await res.json().catch(() => ({}));
   181→   181→        toast.error(data.error ?? "Ошибка генерации документа");
   182→   182→        return;
   183→   183→      }
   184→   184→      const blob = await res.blob();
   185→   185→      const url = URL.createObjectURL(blob);
   186→   186→      const link = document.createElement("a");
   187→   187→      link.href = url;
   188→   188→      const cd = res.headers.get("Content-Disposition") ?? "";
   189→   189→      const m = cd.match(/filename="([^"]+)"/);
   190→   190→      link.download = m?.[1] ?? `document.${target.ext}`;
   191→   191→      document.body.appendChild(link);
   192→   192→      link.click();
   193→   193→      document.body.removeChild(link);
   194→   194→      setTimeout(() => URL.revokeObjectURL(url), 5000);
   195→   195→      toast.success("Документ сформирован", {
   196→   196→        description: target.title,
   197→   197→      });
   198→   198→    } catch (err) {
   199→   199→      console.error(err);
   200→   200→      toast.error("Ошибка генерации документа");
   201→   201→    } finally {
   202→   202→      setGenerating(false);
   203→   203→    }
   204→   204→  }, [profile, selected, generating, targets]);
   205→   205→
   206→   206→  const handleTemplateUpload = useCallback(
   207→   207→    async (file: File) => {
   208→   208→      const targetId = templateTargetRef.current;
   209→   209→      if (!targetId) return;
   210→   210→      setUploadingTemplate(targetId);
   211→   211→      try {
   212→   212→        const fd = new FormData();
   213→   213→        fd.append("file", file);
   214→   214→        fd.append("targetId", targetId);
   215→   215→        const res = await fetch("/api/doc-autofill/templates", {
   216→   216→          method: "POST",
   217→   217→          body: fd,
   218→   218→        });
   219→   219→        const data = await res.json();
   220→   220→        if (!res.ok) {
   221→   221→          toast.error(data.error ?? "Не удалось загрузить шаблон");
   222→   222→          return;
   223→   223→        }
   224→   224→        toast.success("Шаблон загружен");
   225→   225→        await loadTargets();
   226→   226→      } catch {
   227→   227→        toast.error("Не удалось загрузить шаблон");
   228→   228→      } finally {
   229→   229→        setUploadingTemplate(null);
   230→   230→      }
   231→   231→    },
   232→   232→    [loadTargets]
   233→   233→  );
   234→   234→
   235→   235→  const setField = (key: keyof MerchantProfile, value: unknown) => {
   236→   236→    setProfile((p) => (p ? { ...p, [key]: value } : p));
   237→   237→  };
   238→   238→
   239→   239→  // при смене целевого документа сбрасываем закрепления инпутов
   240→   240→  useEffect(() => {
   241→   241→    setPinned(new Set());
   242→   242→  }, [selected]);
   243→   243→
   244→   244→  const filledCount = profile
   245→   245→    ? Object.entries(profile).filter(([k, v]) =>
   246→   246→        LIST_KEYS.has(k)
   247→   247→          ? (v as string[]).length > 0
   248→   248→          : Boolean(String(v ?? "").trim())
   249→   249→      ).length
   250→   250→    : 0;
   251→   251→
   252→   252→  return (
   253→   253→    <div className="flex-1 overflow-y-auto stage-bg">
   254→   254→      <div className="max-w-5xl mx-auto px-4 py-8 md:py-10 w-full">
   255→   255→        {/* Hero */}
   256→   256→        <div className="text-center mb-8 stagger-item">
   257→   257→          <div className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full bg-card/80 border border-border/60 text-xs text-muted-foreground mb-4">
   258→   258→            <Sparkles className="h-3.5 w-3.5 text-terracotta" />
   259→   259→            Данные подставляются по смыслу
   260→   260→          </div>
   261→   261→          <h2 className="display-title text-3xl md:text-4xl">
   262→   262→            Автозаполнение{" "}
   263→   263→            <span className="text-terracotta-dark">документов</span>
   264→   264→          </h2>
   265→   265→          <p className="text-sm text-muted-foreground mt-2 max-w-xl mx-auto">
   266→   266→            Загрузите заполненную анкету мерчанта — данные автоматически
   267→   267→            встанут в заявку на регистрацию СБП или анкету-заявление ИП/ЮЛ.
   268→   268→          </p>
   269→   269→        </div>
   270→   270→
   271→   271→        <div className="space-y-5">
   272→   272→          {/* Шаг 1 — анкета */}
   273→   273→          <section
   274→   274→            className="bg-card border border-border/70 rounded-2xl shadow-paper p-5 md:p-6 stagger-item"
   275→   275→            style={{ ["--stagger-delay" as string]: "60ms" }}
   276→   276→          >
   277→   277→            <div className="flex items-start justify-between gap-3 mb-4">
   278→   278→              <div>
   279→   279→                <div className="text-[11px] uppercase tracking-widest text-muted-foreground font-medium">
   280→   280→                  Шаг 1
   281→   281→                </div>
   282→   282→                <h3 className="display-title text-xl mt-0.5">
   283→   283→                  Заполненная анкета
   284→   284→                </h3>
   285→   285→              </div>
   286→   286→              {profile && (
   287→   287→                <Button
   288→   288→                  variant="outline"
   289→   289→                  size="sm"
   290→   290→                  className="gap-1.5 rounded-xl"
   291→   291→                  onClick={() => anketaInputRef.current?.click()}
   292→   292→                >
   293→   293→                  <RefreshCw className="h-3.5 w-3.5" />
   294→   294→                  Другая
   295→   295→                </Button>
   296→   296→              )}
   297→   297→            </div>
   298→   298→
   299→   299→            {!profile ? (
   300→   300→              <button
   301→   301→                type="button"
   302→   302→                onClick={() => anketaInputRef.current?.click()}
   303→   303→                disabled={parsing}
   304→   304→                className="w-full rounded-xl border-2 border-dashed border-border bg-secondary/30 hover:bg-secondary/50 transition-colors p-8 flex flex-col items-center gap-3 text-center disabled:opacity-60"
   305→   305→              >
   306→   306→                {parsing ? (
   307→   307→                  <Loader2 className="h-8 w-8 text-terracotta animate-spin" />
   308→   308→                ) : (
   309→   309→                  <FileSpreadsheet className="h-8 w-8 text-muted-foreground" />
   310→   310→                )}
   311→   311→                <div>
   312→   312→                  <div className="font-medium text-sm">
   313→   313→                    {parsing
   314→   314→                      ? "Разбираем анкету…"
   315→   315→                      : "Нажмите, чтобы выбрать .xlsx анкету"}
   316→   316→                  </div>
   317→   317→                  <div className="text-xs text-muted-foreground mt-1">
   318→   318→                    «Заявка на подключение» (Uniteller) после заполнения
   319→   319→                  </div>
   320→   320→                </div>
   321→   321→              </button>
   322→   322→            ) : (
   323→   323→              <div className="flex items-center gap-3 p-3.5 rounded-xl bg-oat/60 border border-border/50">
   324→   324→                <div className="h-10 w-10 rounded-xl bg-emerald-100 flex items-center justify-center shrink-0">
   325→   325→                  <Check className="h-5 w-5 text-emerald-700" />
   326→   326→                </div>
   327→   327→                <div className="min-w-0 flex-1">
   328→   328→                  <div className="text-sm font-medium truncate">{sourceName}</div>
   329→   329→                  <div className="text-xs text-muted-foreground">
   330→   330→                    Распознано полей: {filledCount}
   331→   331→                  </div>
   332→   332→                </div>
   333→   333→              </div>
   334→   334→            )}
   335→   335→
   336→   336→            {warnings.length > 0 && profile && (
   337→   337→              <div className="mt-3 rounded-xl bg-amber-50 border border-amber-200/70 p-3 space-y-1.5">
   338→   338→                {warnings.map((w, i) => (
   339→   339→                  <div
   340→   340→                    key={i}
   341→   341→                    className="flex items-start gap-2 text-xs text-amber-800"
   342→   342→                  >
   343→   343→                    <AlertCircle className="h-3.5 w-3.5 mt-0.5 shrink-0" />
   344→   344→                    {w}
   345→   345→                  </div>
   346→   346→                ))}
   347→   347→              </div>
   348→   348→            )}
   349→   349→          </section>
   350→   350→
   351→   351→          {/* Шаг 2 — профиль */}
   352→   352→          {profile && (
   353→   353→            <section
   354→   354→              className="bg-card border border-border/70 rounded-2xl shadow-paper p-5 md:p-6 stagger-item"
   355→   355→              style={{ ["--stagger-delay" as string]: "120ms" }}
   356→   356→            >
   357→   357→              <div className="mb-4">
   358→   358→                <div className="text-[11px] uppercase tracking-widest text-muted-foreground font-medium">
   359→   359→                  Шаг 2
   360→   360→                </div>
   361→   361→                <h3 className="display-title text-xl mt-0.5">
   362→   362→                  Данные мерчанта
   363→   363→                </h3>
   364→   364→                <p className="text-xs text-muted-foreground mt-1">
   365→   365→                  Проверьте и при необходимости поправьте — в документы встанут
   366→   366→                  именно эти значения.
   367→   367→                </p>
   368→   368→              </div>
   369→   369→
   370→   370→              <div className="grid md:grid-cols-2 gap-4">
   371→   371→                {PROFILE_GROUPS.map((group) => (

   374→   374→                    className="rounded-xl border border-border/50 bg-secondary/20 p-4"
   375→   375→                  >
   376→   376→                    <div className="text-xs font-semibold text-terracotta-dark uppercase tracking-wide mb-3">
   377→   377→                      {group.title}
   378→   378→                    </div>
   379→   379→                    <div className="space-y-2.5">
   380→   380→                      {group.fields.map((f) => (

   388→   388→                                "\n"

   391→   391→                                setField(
   392→   392→                                  f.key,
   393→   393→                                  e.target.value

   400→   400→                                4,
   401→   401→                                Math.max(
   402→   402→                                  2,

   404→   404→                                    2
   405→   405→                                )
   406→   406→                              )}
   407→   407→                              placeholder="По одному в строке"
   408→   408→                              className="w-full text-sm rounded-lg border border-border/60 bg-card px-3 py-2 focus:outline-none focus:ring-2 focus:ring-terracotta/30 focus:border-terracotta/50 resize-y min-h-[60px] max-h-48"
   409→   409→                            />
   410→   410→                          ) : (
   411→   411→                            <input
   412→   412→                              value={String(profile[f.key] ?? "")}
   413→   413→                              onChange={(e) =>
   414→   414→                                setField(f.key, e.target.value)
   415→   415→                              }
   416→   416→                              className="w-full text-sm rounded-lg border border-border/60 bg-card px-3 py-2 focus:outline-none focus:ring-2 focus:ring-terracotta/30 focus:border-terracotta/50"
   417→   417→                            />
   418→   418→                          )}
   419→   419→                        </div>
   420→   420→                      ))}
   421→   421→                    </div>
   422→   422→                  </div>
   423→   423→                ))}
   424→   424→              </div>
   425→   425→            </section>
   426→   426→          )}
   427→   427→
   428→   428→          {/* Шаг 3 — целевой документ */}
   429→   429→          {profile && (
   430→   430→            <section
   431→   431→              className="bg-card border border-border/70 rounded-2xl shadow-paper p-5 md:p-6 stagger-item"
   432→   432→              style={{ ["--stagger-delay" as string]: "180ms" }}
   433→   433→            >
   434→   434→              <div className="mb-4">
   435→   435→                <div className="text-[11px] uppercase tracking-widest text-muted-foreground font-medium">
   436→   436→                  Шаг 3
   437→   437→                </div>
   438→   438→                <h3 className="display-title text-xl mt-0.5">
   439→   439→                  Куда подставить данные
   440→   440→                </h3>
   441→   441→              </div>
   442→   442→
   443→   443→              <div className="grid sm:grid-cols-3 gap-3">
   444→   444→                {targets.map((t) => {
   445→   445→                  const active = selected === t.id && t.available;
   446→   446→                  return (
   447→   447→                    <div key={t.id} className="relative">
